```go
// Example payment model
type Payment struct {
    ID              int64      `db:"id"`
    UserID          int64      `db:"user_id"`
    StripePaymentID string     `db:"stripe_payment_id"`
    Amount          float64    `db:"amount"`
    Currency        string     `db:"currency"`
    Status          string     `db:"status"`
    PaymentType     string     `db:"payment_type"`
    CreatedAt       time.Time  `db:"created_at"`
    UpdatedAt       time.Time  `db:"updated_at"`
}

// Example database functions
func (s *PaymentStore) CreatePayment(ctx context.Context, payment *Payment) error {
    query := `
        INSERT INTO payments (user_id, stripe_payment_id, amount, currency, status, payment_type)
        VALUES ($1, $2, $3, $4, $5, $6)
        RETURNING id, created_at, updated_at
    `
    
    row := s.db.QueryRowContext(
        ctx, 
        query, 
        payment.UserID, 
        payment.StripePaymentID, 
        payment.Amount, 
        payment.Currency, 
        payment.Status, 
        payment.PaymentType,
    )
    
    return row.Scan(&payment.ID, &payment.CreatedAt, &payment.UpdatedAt)
}

func (s *PaymentStore) UpdatePaymentStatus(ctx context.Context, paymentID string, status string) error {
    query := `
        UPDATE payments
        SET status = $1, updated_at = NOW()
        WHERE stripe_payment_id = $2
    `
    
    _, err := s.db.ExecContext(ctx, query, status, paymentID)
    return err
}
```

- Implement functions to create and confirm payment intents with Stripe

- Create handlers for payment success and failure callbacks

- Store payment records in the database with appropriate status

```go
// Example payment service
type PaymentService struct {
    stripeService StripeService
    paymentStore  PaymentStore
    logger        Logger
}

func (s *PaymentService) ProcessOneTimePayment(ctx context.Context, userID int64, amount float64, currency string, metadata map[string]string) (*Payment, error) {
    // Create payment record in pending state
    payment := &Payment{
        UserID:      userID,
        Amount:      amount,
        Currency:    currency,
        Status:      "pending",
        PaymentType: "one_time",
    }
    
    // Create payment intent with Stripe
    stripeAmount := int64(amount * 100) // Convert to cents
    pi, err := s.stripeService.CreatePaymentIntent(stripeAmount, currency, metadata)
    if err != nil {
        paymentError := s.stripeService.HandleStripeError(err)
        s.logger.Error("Failed to create payment intent", 
            "error", err,
            "error_type", paymentError.Type,
            "user_message", paymentError.UserMessage)
        return nil, fmt.Errorf("failed to create payment intent: %w", err)
    }
    
    // Update payment record with Stripe ID
    payment.StripePaymentID = pi.ID
    
    // Save payment to database
    if err := s.paymentStore.CreatePayment(ctx, payment); err != nil {
        return nil, fmt.Errorf("failed to save payment record: %w", err)
    }
    
    return payment, nil
}

func (s *PaymentService) ConfirmPayment(ctx context.Context, paymentIntentID string) error {
    // Confirm payment with Stripe
    pi, err := s.stripeService.ConfirmPayment(paymentIntentID)
    if err != nil {
        paymentError := s.stripeService.HandleStripeError(err)
        s.logger.Error("Failed to confirm payment", 
            "error", err,
            "error_type", paymentError.Type,
            "payment_intent_id", paymentIntentID)
            
        // Log the payment error for tracking
        s.stripeService.LogPaymentError("", paymentIntentID, paymentError)
        
        return fmt.Errorf("failed to confirm payment: %w", err)
    }
    
    // Update payment status in database
    status := "succeeded"
    if pi.Status != stripe.PaymentIntentStatusSucceeded {
        status = "failed"
    }
    
    if err := s.paymentStore.UpdatePaymentStatus(ctx, paymentIntentID, status); err != nil {
        s.logger.Error("Failed to update payment status", "error", err, "payment_intent_id", paymentIntentID)
        return fmt.Errorf("failed to update payment status: %w", err)
    }
    
    return nil
}
```

```go
// Example error handling in HTTP handler
func (h *PaymentHandler) HandlePaymentError(w http.ResponseWriter, r *http.Request, err error) {
    paymentError := h.stripeService.HandleStripeError(err)
    
    // Log the error for tracking
    h.logger.Error("Payment error", 
        "error", err, 
        "error_type", paymentError.Type, 
        "error_code", paymentError.Code)
    
    // Render error page with user-friendly message
    data := map[string]interface{}{
        "UserMessage": paymentError.UserMessage,
        "Retryable": paymentError.Retryable,
        "ReferenceID": uuid.New().String(), // For support inquiries
    }
    
    w.WriteHeader(http.StatusBadRequest)
    h.templates.ExecuteTemplate(w, "payment_error.html", data)
}
```

```go
// Example receipt sending
func (s *PaymentService) SendReceipt(ctx context.Context, paymentID string) error {
    // Get payment details
    payment, err := s.paymentStore.GetPaymentByID(ctx, paymentID)
    if err != nil {
        return fmt.Errorf("failed to get payment: %w", err)
    }
    
    // Get user email
    user, err := s.userStore.GetUserByID(ctx, payment.UserID)
    if err != nil {
        return fmt.Errorf("failed to get user: %w", err)
    }
    
    // Create receipt data
    receiptData := map[string]interface{}{
        "PaymentID":   payment.ID,
        "Amount":      payment.Amount,
        "Currency":    payment.Currency,
        "Date":        payment.CreatedAt.Format("January 2, 2006"),
        "PaymentType": payment.PaymentType,
        "UserName":    user.Name,
    }
    
    // Send receipt email
    return s.emailService.SendTemplatedEmail(ctx, user.Email, "Payment Receipt", "payment_receipt", receiptData)
}
```
