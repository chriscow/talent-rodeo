# T005.1.4: Implement Webhook Handler

## Description
Implement a secure webhook handler for processing Stripe events. This includes creating a dedicated endpoint to receive webhook events, implementing signature verification, processing critical payment events, and ensuring idempotent event handling to prevent duplicate processing.

## Implementation Details

### Webhook Endpoint Creation
- Create a dedicated HTTP endpoint to receive Stripe webhook events
- Configure the endpoint in the router/API server
- Implement request body parsing for webhook payloads
- Return immediate 2xx responses before processing events
- Handle errors gracefully without exposing sensitive information

```go
// Example webhook endpoint handler
func (h *StripeWebhookHandler) HandleWebhook(w http.ResponseWriter, r *http.Request) {
    const MaxBodyBytes = int64(65536)
    r.Body = http.MaxBytesReader(w, r.Body, MaxBodyBytes)
    
    payload, err := ioutil.ReadAll(r.Body)
    if err != nil {
        h.logger.Error("Error reading webhook payload", "error", err)
        w.WriteHeader(http.StatusServiceUnavailable)
        return
    }
    
    // Get the signature from the header
    sigHeader := r.Header.Get("Stripe-Signature")
    
    // Verify webhook signature before proceeding
    event, err := webhook.ConstructEvent(payload, sigHeader, h.config.WebhookSecret)
    if err != nil {
        h.logger.Error("Webhook signature verification failed", "error", err)
        w.WriteHeader(http.StatusBadRequest)
        return
    }
    
    // Immediately return a 200 response before processing
    // This prevents timeouts and unnecessary retries from Stripe
    w.WriteHeader(http.StatusOK)
    
    // Process the event asynchronously
    go h.processWebhookEvent(event)
}
```

### Event Processing
- Process events asynchronously to prevent webhook timeout
- Implement idempotent event handling to avoid duplicate processing
- Create handlers for the three critical event types for MVP:
  1. `payment_intent.succeeded`
  2. `payment_intent.payment_failed`
  3. `checkout.session.completed`
- Create a database table to track processed events
- Log all webhook events for debugging and auditing

```go
// Example webhook event processing
func (h *StripeWebhookHandler) processWebhookEvent(event stripe.Event) {
    // Check if we've already processed this event
    processed, err := h.isEventProcessed(event.ID)
    if err != nil {
        h.logger.Error("Error checking event processing status", "error", err, "event_id", event.ID)
        return
    }
    
    if processed {
        h.logger.Info("Skipping already processed event", "event_id", event.ID, "event_type", event.Type)
        return
    }
    
    // Process based on event type
    var handlerErr error
    switch event.Type {
    case "payment_intent.succeeded":
        handlerErr = h.handlePaymentIntentSucceeded(event)
    case "payment_intent.payment_failed":
        handlerErr = h.handlePaymentIntentFailed(event)
    case "checkout.session.completed":
        handlerErr = h.handleCheckoutSessionCompleted(event)
    default:
        h.logger.Info("Ignoring unhandled event type", "event_type", event.Type)
        return
    }
    
    // Record event as processed
    if handlerErr != nil {
        h.logger.Error("Error processing webhook event", 
            "error", handlerErr,
            "event_id", event.ID,
            "event_type", event.Type)
        
        // TODO: Consider implementing a retry mechanism for failed events
        
        // Record event as processed with error
        h.markEventProcessed(event.ID, event.Type, handlerErr.Error())
    } else {
        // Record event as processed successfully
        h.markEventProcessed(event.ID, event.Type, "")
        h.logger.Info("Successfully processed webhook event", 
            "event_id", event.ID,
            "event_type", event.Type)
    }
}
```

### Successful Payment Handling
- Implement handler for `payment_intent.succeeded` events
- Extract customer information from payment metadata
- Record the successful payment in the database
- Update the user's subscription_end_date to 30 days from now
- Update the user's tier to 'mid' tier
- Send confirmation notification if needed

```go
// Example payment_intent.succeeded handler
func (h *StripeWebhookHandler) handlePaymentIntentSucceeded(event stripe.Event) error {
    // Parse the payment intent from the event
    var paymentIntent stripe.PaymentIntent
    if err := json.Unmarshal(event.Data.Raw, &paymentIntent); err != nil {
        return fmt.Errorf("error parsing payment intent: %w", err)
    }
    
    // Extract customer ID from metadata
    candidateID, ok := paymentIntent.Metadata["candidate_id"]
    if !ok {
        return fmt.Errorf("payment intent %s has no candidate_id in metadata", paymentIntent.ID)
    }
    
    // Begin transaction
    tx, err := h.db.BeginTx(context.Background(), nil)
    if err != nil {
        return fmt.Errorf("failed to start transaction: %w", err)
    }
    defer tx.Rollback() // Rollback if not committed
    
    // Record the payment
    if _, err := tx.Exec(
        "INSERT INTO payments (candidate_id, stripe_payment_id, amount, currency, status) VALUES ($1, $2, $3, $4, $5)",
        candidateID,
        paymentIntent.ID,
        float64(paymentIntent.Amount) / 100.0, // Convert from cents
        strings.ToLower(string(paymentIntent.Currency)),
        "succeeded",
    ); err != nil {
        return fmt.Errorf("failed to record payment: %w", err)
    }
    
    // Update subscription_end_date (30 days from now)
    subscriptionEndDate := time.Now().AddDate(0, 0, 30)
    if _, err := tx.Exec(
        "UPDATE candidates SET tier = 'mid', subscription_end_date = $1 WHERE id = $2",
        subscriptionEndDate,
        candidateID,
    ); err != nil {
        return fmt.Errorf("failed to update subscription: %w", err)
    }
    
    // Commit transaction
    if err := tx.Commit(); err != nil {
        return fmt.Errorf("failed to commit transaction: %w", err)
    }
    
    return nil
}
```

```go
// Example payment_intent.payment_failed handler
func (h *StripeWebhookHandler) handlePaymentIntentFailed(event stripe.Event) error {
    // Parse the payment intent from the event
    var paymentIntent stripe.PaymentIntent
    if err := json.Unmarshal(event.Data.Raw, &paymentIntent); err != nil {
        return fmt.Errorf("error parsing payment intent: %w", err)
    }
    
    // Extract customer ID from metadata
    candidateID, ok := paymentIntent.Metadata["candidate_id"]
    if !ok {
        return fmt.Errorf("payment intent %s has no candidate_id in metadata", paymentIntent.ID)
    }
    
    // Get the error message
    errorMessage := "Unknown error"
    errorCode := ""
    if paymentIntent.LastPaymentError != nil {
        errorMessage = paymentIntent.LastPaymentError.Message
        errorCode = paymentIntent.LastPaymentError.Code
    }
    
    // Begin transaction
    tx, err := h.db.BeginTx(context.Background(), nil)
    if err != nil {
        return fmt.Errorf("failed to start transaction: %w", err)
    }
    defer tx.Rollback() // Rollback if not committed
    
    // Record the failed payment
    if _, err := tx.Exec(
        "INSERT INTO payments (candidate_id, stripe_payment_id, amount, currency, status) VALUES ($1, $2, $3, $4, $5)",
        candidateID,
        paymentIntent.ID,
        float64(paymentIntent.Amount) / 100.0,
        strings.ToLower(string(paymentIntent.Currency)),
        "failed",
    ); err != nil {
        return fmt.Errorf("failed to record payment: %w", err)
    }
    
    // Log the payment error
    if _, err := tx.Exec(
        "INSERT INTO payment_errors (candidate_id, stripe_payment_id, error_type, error_code, error_message) VALUES ($1, $2, $3, $4, $5)",
        candidateID,
        paymentIntent.ID,
        "payment_failed",
        errorCode,
        errorMessage,
    ); err != nil {
        return fmt.Errorf("failed to log payment error: %w", err)
    }
    
    // Commit transaction
    if err := tx.Commit(); err != nil {
        return fmt.Errorf("failed to commit transaction: %w", err)
    }
    
    return nil
}
```

### Database Implementation
- Create a table for tracking processed webhook events
- Implement functions to check if an event has been processed
- Implement functions to mark events as processed
- Ensure proper indexing for efficient lookups

```go
// Example database functions for event processing
func (h *StripeWebhookHandler) isEventProcessed(eventID string) (bool, error) {
    var count int
    err := h.db.QueryRow("SELECT COUNT(*) FROM stripe_events WHERE id = $1", eventID).Scan(&count)
    if err != nil {
        return false, err
    }
    return count > 0, nil
}

func (h *StripeWebhookHandler) markEventProcessed(eventID, eventType, errorMessage string) error {
    _, err := h.db.Exec(
        "INSERT INTO stripe_events (id, type, error_message) VALUES ($1, $2, $3) ON CONFLICT (id) DO NOTHING",
        eventID,
        eventType,
        errorMessage,
    )
    return err
}
```
