# T005.2.1: Implement Payment Status Tracking

## Description
Implement a system to track and manage payment statuses for user transactions. This includes defining payment status states, creating a service to track payment history, and implementing APIs for other parts of the system to check payment status. The implementation should provide a reliable way to determine a user's payment status for access control and reporting purposes.

## Implementation Details

- Create constants for status values to ensure consistency

```go
// Example payment status constants
const (
    PaymentStatusPending   = "pending"
    PaymentStatusSucceeded = "succeeded"
    PaymentStatusFailed    = "failed"
    PaymentStatusRefunded  = "refunded"
)

// Example validation function
func IsValidPaymentStatus(status string) bool {
    validStatuses := []string{
        PaymentStatusPending,
        PaymentStatusSucceeded,
        PaymentStatusFailed,
        PaymentStatusRefunded,
    }
    
    for _, s := range validStatuses {
        if s == status {
            return true
        }
    }
    
    return false
}

func IsValidStatusTransition(from, to string) bool {
    // Define valid transitions
    transitions := map[string][]string{
        PaymentStatusPending: {
            PaymentStatusSucceeded,
            PaymentStatusFailed,
        },
        PaymentStatusSucceeded: {
            PaymentStatusRefunded,
        },
        PaymentStatusFailed: {
            PaymentStatusPending, // Retry
        },
        PaymentStatusRefunded: {},
    }
    
    allowedTransitions, exists := transitions[from]
    if !exists {
        return false
    }
    
    for _, s := range allowedTransitions {
        if s == to {
            return true
        }
    }
    
    return false
}
```

### Payment Status Service
- Create a service to manage payment status operations
- Implement functions to get current payment status for a user
- Implement functions to update payment status
- Create methods to query payment history
- Add logging for payment status changes

```go
// Example payment status service
type PaymentStatusService struct {
    paymentStore PaymentStore
    logger       Logger
}

func NewPaymentStatusService(paymentStore PaymentStore, logger Logger) *PaymentStatusService {
    return &PaymentStatusService{
        paymentStore: paymentStore,
        logger:       logger,
    }
}

func (s *PaymentStatusService) GetUserPaymentStatus(ctx context.Context, userID int64) (string, error) {
    // Get the most recent payment for the user
    payment, err := s.paymentStore.GetLatestPaymentForUser(ctx, userID)
    if err != nil {
        return "", fmt.Errorf("failed to get latest payment: %w", err)
    }
    
    if payment == nil {
        return "", nil // No payments found
    }
    
    return payment.Status, nil
}

func (s *PaymentStatusService) UpdatePaymentStatus(ctx context.Context, paymentID string, status string) error {
    // Validate the status
    if !IsValidPaymentStatus(status) {
        return fmt.Errorf("invalid payment status: %s", status)
    }
    
    // Get current payment to check transition
    payment, err := s.paymentStore.GetPaymentByStripeID(ctx, paymentID)
    if err != nil {
        return fmt.Errorf("failed to get payment: %w", err)
    }
    
    // Validate the transition
    if !IsValidStatusTransition(payment.Status, status) {
        return fmt.Errorf("invalid status transition from %s to %s", payment.Status, status)
    }
    
    // Update the payment status
    if err := s.paymentStore.UpdatePaymentStatus(ctx, paymentID, status); err != nil {
        return fmt.Errorf("failed to update payment status: %w", err)
    }
    
    s.logger.Info("Updated payment status", 
        "payment_id", paymentID, 
        "old_status", payment.Status, 
        "new_status", status,
        "user_id", payment.UserID)
    
    return nil
}

func (s *PaymentStatusService) GetPaymentHistory(ctx context.Context, userID int64, limit, offset int) ([]*Payment, error) {
    payments, err := s.paymentStore.GetPaymentsForUser(ctx, userID, limit, offset)
    if err != nil {
        return nil, fmt.Errorf("failed to get payment history: %w", err)
    }
    
    return payments, nil
}
```

### Webhook Integration
- Implement webhook handlers to update payment status based on Stripe events
- Handle events for payment intent successes, failures, and refunds
- Add reconciliation logic to ensure database status matches Stripe
- Implement error handling and retry logic for webhook processing

```go
// Example webhook handler for payment status events
func (h *WebhookHandler) HandlePaymentStatusEvent(event stripe.Event) error {
    ctx := context.Background()
    
    switch event.Type {
    case "payment_intent.succeeded":
        var paymentIntent stripe.PaymentIntent
        if err := json.Unmarshal(event.Data.Raw, &paymentIntent); err != nil {
            return fmt.Errorf("failed to parse payment intent: %w", err)
        }
        
        return h.paymentStatusService.UpdatePaymentStatus(ctx, paymentIntent.ID, PaymentStatusSucceeded)
        
    case "payment_intent.payment_failed":
        var paymentIntent stripe.PaymentIntent
        if err := json.Unmarshal(event.Data.Raw, &paymentIntent); err != nil {
            return fmt.Errorf("failed to parse payment intent: %w", err)
        }
        
        return h.paymentStatusService.UpdatePaymentStatus(ctx, paymentIntent.ID, PaymentStatusFailed)
        
    case "charge.refunded":
        var charge stripe.Charge
        if err := json.Unmarshal(event.Data.Raw, &charge); err != nil {
            return fmt.Errorf("failed to parse charge: %w", err)
        }
        
        // Find payment intent ID from charge
        payment, err := h.paymentStore.GetPaymentByChargeID(ctx, charge.ID)
        if err != nil {
            return fmt.Errorf("failed to find payment for charge: %w", err)
        }
        
        return h.paymentStatusService.UpdatePaymentStatus(ctx, payment.StripePaymentID, PaymentStatusRefunded)
        
    default:
        h.logger.Info("Ignoring unhandled payment event", "type", event.Type)
        return nil
    }
}
```

```go
// Example payment status check middleware
func RequirePaymentStatus(requiredStatus string) gin.HandlerFunc {
    return func(c *gin.Context) {
        userID := getUserIDFromContext(c)
        paymentStatusService := c.MustGet("paymentStatusService").(*PaymentStatusService)
        
        status, err := paymentStatusService.GetUserPaymentStatus(c.Request.Context(), userID)
        if err != nil {
            c.JSON(http.StatusInternalServerError, gin.H{"error": "Could not check payment status"})
            c.Abort()
            return
        }
        
        if status != requiredStatus {
            c.JSON(http.StatusForbidden, gin.H{
                "error": "Payment required",
                "current_status": status,
                "required_status": requiredStatus,
                "payment_url": "/payment",
            })
            c.Abort()
            return
        }
        
        c.Next()
    }
}
```

// Example implementation of payment status check API
func (h *PaymentStatusHandler) GetPaymentStatusHandler(c *gin.Context) {
    userID := getUserIDFromContext(c)
    
    status, err := h.paymentStatusService.GetUserPaymentStatus(c.Request.Context(), userID)
    if err != nil {
        h.logger.Error("Failed to get payment status", "error", err)
        c.JSON(http.StatusInternalServerError, gin.H{"error": "Failed to get payment status"})
        return
    }
    
    // Get payment history
    payments, err := h.paymentStatusService.GetPaymentHistory(c.Request.Context(), userID, 10, 0)
    if err != nil {
        h.logger.Error("Failed to get payment history", "error", err)
        // Continue anyway, not critical
        payments = []*Payment{}
    }
    
    // Format payment history for response
    history := make([]map[string]any, 0, len(payments))
    for _, p := range payments {
        history = append(history, map[string]any{
            "id":         p.ID,
            "amount":     p.Amount,
            "currency":   p.Currency,
            "status":     p.Status,
            "created_at": p.CreatedAt.Format(time.RFC3339),
        })
    }
    
    c.JSON(http.StatusOK, gin.H{
        "current_status": status,
        "has_paid": status == PaymentStatusSucceeded,
        "payment_history": history,
    })
}
