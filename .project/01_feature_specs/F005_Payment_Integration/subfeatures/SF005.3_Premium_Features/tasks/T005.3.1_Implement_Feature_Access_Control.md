# T005.3.1: Implement Feature Access Control

## Description
Implement a system to control access to premium features based on user payment status. This includes creating a middleware component to verify access rights, implementing a feature flags system to define which features are premium, and creating a database schema to store feature definitions. The implementation should provide a secure and flexible way to manage access to premium features.

## Implementation Details

```go
func (s *FeatureStore) GetFeatureByKey(ctx context.Context, featureKey string) (*PremiumFeature, error) {
    query := `
        SELECT id, feature_key, display_name, description, is_active, tier_level, created_at, updated_at
        FROM premium_features
        WHERE feature_key = $1
    `
    
    feature := &PremiumFeature{}
    err := s.db.GetContext(ctx, feature, query, featureKey)
    if err != nil {
        if err == sql.ErrNoRows {
            return nil, fmt.Errorf("feature not found: %s", featureKey)
        }
        return nil, err
    }
    
    return feature, nil
}

func (s *FeatureStore) GetFeaturesForTier(ctx context.Context, tierID int64) ([]*PremiumFeature, error) {
    query := `
        SELECT f.id, f.feature_key, f.display_name, f.description, f.is_active, f.tier_level, f.created_at, f.updated_at
        FROM premium_features f
        JOIN tier_features tf ON f.id = tf.feature_id
        WHERE tf.tier_id = $1 AND f.is_active = true
    `
    
    var features []*PremiumFeature
    err := s.db.SelectContext(ctx, &features, query, tierID)
    if err != nil {
        return nil, err
    }
    
    return features, nil
}
```

### Feature Access Service
- Create a service to check user access to premium features
- Implement functions to get all accessible features for a user
- Add caching for frequently accessed feature information
- Create methods to check specific feature access rights
- Support time-limited feature access

```go
// Example feature access service
type FeatureAccessService struct {
    featureStore      FeatureStore
    subscriptionStore SubscriptionStore
    cache             Cache
    logger            Logger
}

func NewFeatureAccessService(featureStore FeatureStore, subscriptionStore SubscriptionStore, cache Cache, logger Logger) *FeatureAccessService {
    return &FeatureAccessService{
        featureStore:      featureStore,
        subscriptionStore: subscriptionStore,
        cache:             cache,
        logger:            logger,
    }
}

func (s *FeatureAccessService) IsFeatureEnabled(ctx context.Context, featureKey string) (bool, error) {
    // Check if feature exists and is active
    cacheKey := fmt.Sprintf("feature_enabled:%s", featureKey)
    
    // Try to get from cache first
    if enabled, found := s.cache.Get(cacheKey); found {
        return enabled.(bool), nil
    }
    
    // Get from database
    feature, err := s.featureStore.GetFeatureByKey(ctx, featureKey)
    if err != nil {
        return false, fmt.Errorf("failed to get feature: %w", err)
    }
    
    enabled := feature.IsActive
    
    // Cache the result
    s.cache.Set(cacheKey, enabled, 5*time.Minute)
    
    return enabled, nil
}

func (s *FeatureAccessService) CheckUserAccess(ctx context.Context, userID int64, featureKey string) (bool, error) {
    // Check if feature is enabled first
    enabled, err := s.IsFeatureEnabled(ctx, featureKey)
    if err != nil {
        return false, err
    }
    
    if !enabled {
        return false, nil
    }
    
    // Cache key for user feature access
    cacheKey := fmt.Sprintf("user_feature_access:%d:%s", userID, featureKey)
    
    // Try to get from cache first
    if hasAccess, found := s.cache.Get(cacheKey); found {
        return hasAccess.(bool), nil
    }
    
    // Get feature tier level
    feature, err := s.featureStore.GetFeatureByKey(ctx, featureKey)
    if err != nil {
        return false, fmt.Errorf("failed to get feature: %w", err)
    }
    
    // Get user's active subscription
    subscription, err := s.subscriptionStore.GetActiveSubscriptionForUser(ctx, userID)
    if err != nil {
        return false, fmt.Errorf("failed to get subscription: %w", err)
    }
    
    // If no subscription, no access
    if subscription == nil {
        s.cache.Set(cacheKey, false, 5*time.Minute)
        return false, nil
    }
    
    // Get subscription tier
    tier, err := s.subscriptionStore.GetTierByID(ctx, subscription.TierID)
    if err != nil {
        return false, fmt.Errorf("failed to get tier: %w", err)
    }
    
    // Check if subscription includes this feature
    tierFeatures, err := s.featureStore.GetFeaturesForTier(ctx, tier.ID)
    if err != nil {
        return false, fmt.Errorf("failed to get tier features: %w", err)
    }
    
    hasAccess := false
    for _, tf := range tierFeatures {
        if tf.FeatureKey == featureKey {
            hasAccess = true
            break
        }
    }
    
    // Cache the result
    s.cache.Set(cacheKey, hasAccess, 5*time.Minute)
    
    return hasAccess, nil
}

func (s *FeatureAccessService) GetUserAccessibleFeatures(ctx context.Context, userID int64) ([]*PremiumFeature, error) {
    // Get user's active subscription
    subscription, err := s.subscriptionStore.GetActiveSubscriptionForUser(ctx, userID)
    if err != nil {
        return nil, fmt.Errorf("failed to get subscription: %w", err)
    }
    
    // If no subscription, no premium features
    if subscription == nil {
        return []*PremiumFeature{}, nil
    }
    
    // Get tier features
    features, err := s.featureStore.GetFeaturesForTier(ctx, subscription.TierID)
    if err != nil {
        return nil, fmt.Errorf("failed to get tier features: %w", err)
    }
    
    return features, nil
}
```

### Access Control Middleware
- Create middleware for checking premium feature access
- Implement route protection based on feature access
- Add support for graceful permission denial
- Implement logging for access attempts
- Support fallbacks for non-premium users

```go
// Example middleware for premium feature access control
func RequirePremiumFeature(featureKey string) gin.HandlerFunc {
    return func(c *gin.Context) {
        userID := getUserIDFromContext(c)
        featureService := c.MustGet("featureService").(*FeatureAccessService)
        logger := c.MustGet("logger").(Logger)
        
        // Check if user has access to the feature
        hasAccess, err := featureService.CheckUserAccess(c.Request.Context(), userID, featureKey)
        if err != nil {
            logger.Error("Failed to check feature access", "error", err, "feature", featureKey, "user_id", userID)
            c.JSON(http.StatusInternalServerError, gin.H{"error": "Could not check feature access"})
            c.Abort()
            return
        }
        
        // Log access attempt
        logger.Info("Premium feature access attempt", 
            "feature", featureKey, 
            "user_id", userID, 
            "granted", hasAccess)
        
        if !hasAccess {
            // Get feature details for the response
            feature, err := featureService.GetFeatureDetails(c.Request.Context(), featureKey)
            if err != nil {
                feature = &PremiumFeature{
                    FeatureKey:  featureKey,
                    DisplayName: "Premium Feature",
                    Description: "This is a premium feature that requires a subscription.",
                }
            }
            
            c.JSON(http.StatusForbidden, gin.H{
                "error": "Premium feature access required",
                "feature": feature.DisplayName,
                "description": feature.Description,
                "upgrade_url": "/pricing",
            })
            c.Abort()
            return
        }
        
        c.Next()
    }
}
```
