# T004.3.1: Raw Skill Scoring

## Task Description
Implement the raw skill scoring component of the match algorithm that assigns points to job listings based on how well they match a candidate's skills. This task includes identifying skill matches between candidate skills and job requirements and assigning appropriate point values.

## Acceptance Criteria
1. The function correctly identifies matches between candidate skills and job requirements
2. "Must Have" skill matches are awarded 10 points each
3. "Preferred" skill matches are awarded 5 points each
4. Each skill match is recorded in the `skill_matches` table
5. The total raw score is calculated and stored in the `job_matches` table
6. The function handles edge cases gracefully (no skills, no matches, etc.)

## Implementation Details
1. Update the `scoreJobs` function in `/internal/handlers/search.go`:
   ```go
   // scoreJobs calculates a raw match score for each job based on skill matches
   func scoreJobs(jobs []models.Job, candidateSkills []models.CandidateSkill, app *config.Application) ([]ScoredJob, error) {
       // Return early if no jobs or no skills
       if len(jobs) == 0 || len(candidateSkills) == 0 {
           return []ScoredJob{}, nil
       }
       
       // Organize candidate skills by ID for easier access
       skillMap := make(map[int]models.CandidateSkill)
       mustHaveSkills := []models.CandidateSkill{}
       preferredSkills := []models.CandidateSkill{}
       
       for _, skill := range candidateSkills {
           skillMap[skill.ID] = skill
           if skill.IsMustHave {
               mustHaveSkills = append(mustHaveSkills, skill)
           } else {
               preferredSkills = append(preferredSkills, skill)
           }
       }
       
       // Score each job
       scoredJobs := make([]ScoredJob, 0, len(jobs))
       for _, job := range jobs {
           // Extract job skills from job description
           jobSkills, err := extractJobSkills(job.Description)
           if err != nil {
               log.Printf("Error extracting skills for job %d: %v", job.ID, err)
               continue
           }
           
           // Calculate raw score
           rawScore := 0
           skillMatches := []*models.SkillMatch{}
           
           // Check each candidate skill against job skills
           for _, candidateSkill := range candidateSkills {
               isMatch := false
               confidence := 0.0
               
               // Check if the skill exists in job skills
               for _, jobSkill := range jobSkills {
                   // Simple exact match for MVP
                   if strings.EqualFold(candidateSkill.Name, jobSkill) {
                       isMatch = true
                       confidence = 1.0
                       
                       // Add points based on skill importance
                       if candidateSkill.IsMustHave {
                           rawScore += 10
                       } else {
                           rawScore += 5
                       }
                       
                       break
                   }
               }
               
               // Record skill match
               skillMatches = append(skillMatches, &models.SkillMatch{
                   SkillID:         candidateSkill.ID,
                   IsMatch:         isMatch,
                   MatchConfidence: confidence,
               })
           }
           
           // Create job match record
           jobMatch := &models.JobMatch{
               UserID:   candidateSkills[0].UserID, // Assuming all skills have same user ID
               JobID:    job.ID,
               RawScore: rawScore,
           }
           
           // Add to scored jobs list
           scoredJobs = append(scoredJobs, ScoredJob{
               Job:        job,
               RawScore:   rawScore,
               JobMatch:   jobMatch,
               SkillMatches: skillMatches,
           })
       }
       
       // Store matches in database
       if err := storeJobMatches(scoredJobs, app); err != nil {
           return nil, fmt.Errorf("error storing job matches: %w", err)
       }
       
       return scoredJobs, nil
   }
   
   // ScoredJob combines a job with its score and match details
   type ScoredJob struct {
       Job          models.Job
       RawScore     int
       Percentage   int
       JobMatch     *models.JobMatch
       SkillMatches []*models.SkillMatch
   }
   
   // storeJobMatches persists job matches and skill matches to the database
   func storeJobMatches(scoredJobs []ScoredJob, app *config.Application) error {
       for i := range scoredJobs {
           // Insert job match first to get ID
           err := app.JobMatches.Insert(scoredJobs[i].JobMatch)
           if err != nil {
               return err
           }
           
           // Update skill matches with job match ID
           for j := range scoredJobs[i].SkillMatches {
               scoredJobs[i].SkillMatches[j].JobMatchID = scoredJobs[i].JobMatch.ID
           }
           
           // Insert skill matches in a batch
           err = app.SkillMatches.InsertBatch(scoredJobs[i].SkillMatches)
           if err != nil {
               return err
           }
       }
       
       return nil
   }
   
   // extractJobSkills extracts skills from job description (simplified version for MVP)
   func extractJobSkills(description string) ([]string, error) {
       // For MVP, this will be a simple keyword extraction
       // In a real implementation, this would use the results from F003a Job Data Extraction
       
       // Convert to lowercase for case-insensitive matching
       descLower := strings.ToLower(description)
       
       // Common technical skills to look for
       commonSkills := []string{
           "python", "javascript", "java", "go", "golang", "c#", "ruby",
           "react", "angular", "vue", "node.js", "typescript",
           "aws", "azure", "gcp", "docker", "kubernetes",
           "sql", "nosql", "mongodb", "postgres", "mysql",
           "machine learning", "data science", "ai", "devops", "ci/cd",
       }
       
       // Extract skills that appear in the description
       extractedSkills := []string{}
       for _, skill := range commonSkills {
           if strings.Contains(descLower, strings.ToLower(skill)) {
               extractedSkills = append(extractedSkills, skill)
           }
       }
       
       return extractedSkills, nil
   }
   ```
