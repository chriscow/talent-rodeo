# T004.2.1: Job Matches Table

## Task Description
Implement the `job_matches` table in the database schema to store job match results, scores, LLM assessments, and user feedback. This table consolidates data that was previously planned for multiple tables into a single, streamlined structure.

## Acceptance Criteria
1. The `job_matches` table is created in the database with all required fields
2. Appropriate indices are added for optimized querying
3. Foreign key constraints are properly implemented
4. Data access methods are created for CRUD operations
5. Existing migration files are updated rather than creating new ones

## Implementation Details
1. Modify the existing migration file at `/workspaces/talent-rodeo/assets/migrations/000011_add_match_assessments_table.up.sql`:
   ```sql
   -- Drop the existing match_assessments table if it exists
   DROP TABLE IF EXISTS match_assessments;
   
   -- Create the new job_matches table
   CREATE TABLE job_matches (
       id SERIAL PRIMARY KEY,
       user_id INTEGER NOT NULL REFERENCES users(id),
       job_id INTEGER NOT NULL REFERENCES jobs(id),
       raw_score INTEGER,
       match_percentage INTEGER,
       llm_rating INTEGER,
       llm_explanation TEXT,
       user_feedback TEXT,
       user_feedback_category VARCHAR(50),
       created_at TIMESTAMP DEFAULT NOW(),
   ```

3. Create a new JobMatchModel in `/internal/models/job_match.go`:
   ```go
   package models

   import (
       "database/sql"
       "time"
   )

   // JobMatch represents a match between a user and a job posting
   type JobMatch struct {
       ID                  int
       UserID              int
       JobID               int
       RawScore            int
       MatchPercentage     int
       LLMRating           int
       LLMExplanation      string
       UserFeedback        string
       UserFeedbackCategory string
       CreatedAt           time.Time
       UpdatedAt           time.Time
   }

   // JobMatchModel wraps a database connection pool
   type JobMatchModel struct {
       DB *sql.DB
   }

   // Insert adds a new job match to the database
   func (m *JobMatchModel) Insert(match *JobMatch) error {
       stmt := `
           INSERT INTO job_matches 
               (user_id, job_id, raw_score, match_percentage, llm_rating, 
                llm_explanation, user_feedback, user_feedback_category)
           VALUES ($1, $2, $3, $4, $5, $6, $7, $8)
           RETURNING id, created_at, updated_at
       `

       return m.DB.QueryRow(
           stmt, 
           match.UserID, 
           match.JobID,
           match.RawScore,
           match.MatchPercentage,
           match.LLMRating,
           match.LLMExplanation,
           match.UserFeedback,
           match.UserFeedbackCategory,
       ).Scan(&match.ID, &match.CreatedAt, &match.UpdatedAt)
   }

   // Get retrieves a specific job match by ID
   func (m *JobMatchModel) Get(id int) (*JobMatch, error) {
       match := &JobMatch{}
       
       stmt := `
           SELECT id, user_id, job_id, raw_score, match_percentage, 
                  llm_rating, llm_explanation, user_feedback, 
                  user_feedback_category, created_at, updated_at
           FROM job_matches
           WHERE id = $1
       `
       
       err := m.DB.QueryRow(stmt, id).Scan(
           &match.ID,
           &match.UserID,
           &match.JobID,
           &match.RawScore,
           &match.MatchPercentage,
           &match.LLMRating,
           &match.LLMExplanation,
           &match.UserFeedback,
           &match.UserFeedbackCategory,
           &match.CreatedAt,
           &match.UpdatedAt,
       )
       
       if err != nil {
           return nil, err
       }
       
       return match, nil
   }

   // GetMatchesForUser retrieves all matches for a specific user
   func (m *JobMatchModel) GetMatchesForUser(userID int) ([]*JobMatch, error) {
       stmt := `
           SELECT id, user_id, job_id, raw_score, match_percentage, 
                  llm_rating, llm_explanation, user_feedback, 
                  user_feedback_category, created_at, updated_at
           FROM job_matches
           WHERE user_id = $1
           ORDER BY match_percentage DESC
       `
       
       rows, err := m.DB.Query(stmt, userID)
       if err != nil {
           return nil, err
       }
       defer rows.Close()
       
       matches := []*JobMatch{}
       
       for rows.Next() {
           match := &JobMatch{}
           err := rows.Scan(
               &match.ID,
               &match.UserID,
               &match.JobID,
               &match.RawScore,
               &match.MatchPercentage,
               &match.LLMRating,
               &match.LLMExplanation,
               &match.UserFeedback,
               &match.UserFeedbackCategory,
               &match.CreatedAt,
               &match.UpdatedAt,
           )
           
           if err != nil {
               return nil, err
           }
           
           matches = append(matches, match)
       }
       
       if err = rows.Err(); err != nil {
           return nil, err
       }
       
       return matches, nil
   }

   // UpdateFeedback updates user feedback for a specific match
   func (m *JobMatchModel) UpdateFeedback(id int, feedback string, category string) error {
       stmt := `
           UPDATE job_matches
           SET user_feedback = $1, 
               user_feedback_category = $2,
               updated_at = NOW()
           WHERE id = $3
       `
       
       _, err := m.DB.Exec(stmt, feedback, category, id)
       return err
   }
   ```
