# T004.2.2: Skill Matches Table

## Task Description
Implement the `skill_matches` table in the database schema to store detailed information about which candidate skills match with job requirements. This table will track individual skill matches that contribute to the overall job match score.

## Acceptance Criteria
1. The `skill_matches` table is created in the database with all required fields
2. Appropriate indices are added for optimized querying
3. Foreign key constraints are properly implemented
4. Data access methods are created for CRUD operations
5. Existing migration files are updated rather than creating new ones

## Implementation Details
1. Add the `skill_matches` table to the existing migration file at `/workspaces/talent-rodeo/assets/migrations/000011_add_match_assessments_table.up.sql` (after the job_matches table creation):
   ```sql
   -- Create the skill_matches table
   CREATE TABLE skill_matches (
       id SERIAL PRIMARY KEY,
       job_match_id INTEGER NOT NULL REFERENCES job_matches(id) ON DELETE CASCADE,
       skill_id INTEGER NOT NULL REFERENCES candidate_skills(id),
       is_match BOOLEAN NOT NULL DEFAULT false,
       match_confidence FLOAT NOT NULL DEFAULT 0.0,
       created_at TIMESTAMP DEFAULT NOW()
   );
   
   -- Add indices for common query patterns
   CREATE INDEX idx_skill_matches_job_match_id ON skill_matches(job_match_id);
   CREATE INDEX idx_skill_matches_skill_id ON skill_matches(skill_id);
   CREATE UNIQUE INDEX idx_skill_matches_job_match_skill ON skill_matches(job_match_id, skill_id);
   ```

3. Create a new SkillMatchModel in `/internal/models/skill_match.go`:
   ```go
   package models

   import (
       "database/sql"
       "time"
   )

   // SkillMatch represents a match between a candidate skill and a job requirement
   type SkillMatch struct {
       ID              int
       JobMatchID      int
       SkillID         int
       IsMatch         bool
       MatchConfidence float64
       CreatedAt       time.Time
   }

   // SkillMatchModel wraps a database connection pool
   type SkillMatchModel struct {
       DB *sql.DB
   }

   // Insert adds a new skill match to the database
   func (m *SkillMatchModel) Insert(match *SkillMatch) error {
       stmt := `
           INSERT INTO skill_matches 
               (job_match_id, skill_id, is_match, match_confidence)
           VALUES ($1, $2, $3, $4)
           RETURNING id, created_at
       `

       return m.DB.QueryRow(
           stmt, 
           match.JobMatchID, 
           match.SkillID,
           match.IsMatch,
           match.MatchConfidence,
       ).Scan(&match.ID, &match.CreatedAt)
   }

   // InsertBatch adds multiple skill matches to the database in a single transaction
   func (m *SkillMatchModel) InsertBatch(matches []*SkillMatch) error {
       tx, err := m.DB.Begin()
       if err != nil {
           return err
       }
       
       stmt := `
           INSERT INTO skill_matches 
               (job_match_id, skill_id, is_match, match_confidence)
           VALUES ($1, $2, $3, $4)
       `
       
       for _, match := range matches {
           _, err := tx.Exec(
               stmt,
               match.JobMatchID,
               match.SkillID,
               match.IsMatch,
               match.MatchConfidence,
           )
           
           if err != nil {
               tx.Rollback()
               return err
           }
       }
       
       return tx.Commit()
   }

   // GetMatchesForJobMatch retrieves all skill matches for a specific job match
   func (m *SkillMatchModel) GetMatchesForJobMatch(jobMatchID int) ([]*SkillMatch, error) {
       stmt := `
           SELECT id, job_match_id, skill_id, is_match, match_confidence, created_at
           FROM skill_matches
           WHERE job_match_id = $1
       `
       
       rows, err := m.DB.Query(stmt, jobMatchID)
       if err != nil {
           return nil, err
       }
       defer rows.Close()
       
       matches := []*SkillMatch{}
       
       for rows.Next() {
           match := &SkillMatch{}
           err := rows.Scan(
               &match.ID,
               &match.JobMatchID,
               &match.SkillID,
               &match.IsMatch,
               &match.MatchConfidence,
               &match.CreatedAt,
           )
           
           if err != nil {
               return nil, err
           }
           
           matches = append(matches, match)
       }
       
       if err = rows.Err(); err != nil {
           return nil, err
       }
       
       return matches, nil
   }

   // GetMatchingSkillCount returns the count of matching skills for a job match
   func (m *SkillMatchModel) GetMatchingSkillCount(jobMatchID int) (int, error) {
       var count int
       
       stmt := `
           SELECT COUNT(*) 
           FROM skill_matches
           WHERE job_match_id = $1 AND is_match = true
       `
       
       err := m.DB.QueryRow(stmt, jobMatchID).Scan(&count)
       return count, err
   }
   ```

4. Update the `JobMatchModel` to include a method for retrieving matches with their skills:
   ```go
   // GetMatchWithSkills retrieves a job match with its associated skill matches
   func (m *JobMatchModel) GetMatchWithSkills(id int, skillModel *SkillMatchModel) (*JobMatchWithSkills, error) {
       match, err := m.Get(id)
       if err != nil {
           return nil, err
       }
       
       skillMatches, err := skillModel.GetMatchesForJobMatch(id)
       if err != nil {
           return nil, err
       }
       
       return &JobMatchWithSkills{
           JobMatch:     *match,
           SkillMatches: skillMatches,
       }, nil
   }
   
   // JobMatchWithSkills combines a job match with its skill matches
   type JobMatchWithSkills struct {
       JobMatch     JobMatch
       SkillMatches []*SkillMatch
   }
   ```

## Technical Notes
- The `skill_matches` table is designed to track individual skill matches that contribute to the overall job match score
- The `ON DELETE CASCADE` constraint ensures that skill matches are automatically deleted when a job match is deleted
- The batch insert method is provided for efficiently adding multiple skill matches at once
- Consider adding additional indices if query patterns reveal performance bottlenecks
