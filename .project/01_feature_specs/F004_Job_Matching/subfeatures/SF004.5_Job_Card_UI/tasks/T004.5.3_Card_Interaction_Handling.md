# T004.5.3: Card Interaction Handling

## Task Description
Implement client-side and server-side functionality for user interactions with job cards, including the "Interested" and "Not Interested" buttons, feedback collection, and smooth transition animations when users provide feedback.

## Acceptance Criteria
1. Clicking "Interested" or "Not Interested" triggers the appropriate action
2. When a user indicates they're not interested, the feedback dropdown appears
3. Feedback is properly collected and stored in the database
4. Cards smoothly fade out when a user provides feedback
5. New cards automatically load to replace those that have been acted on
6. Form submissions are handled asynchronously without page reloads
7. Error states are handled gracefully with user feedback

## Implementation Details
1. Create a JavaScript file for job card interactions at `/static/js/job-cards.js`:
   ```javascript
   document.addEventListener('DOMContentLoaded', function() {
       // Initialize job cards interactions
       initJobCardInteractions();
       
       // Setup vote and feedback handling
       setupVoteHandlers();
   });
   
   function initJobCardInteractions() {
       // Initialize any card interactions, hover effects, etc.
       console.log('Job card interactions initialized');
   }
   
   function setupVoteHandlers() {
       // Select all action buttons
       const actionButtons = document.querySelectorAll('.interested-btn, .not-interested-btn');
       
       // Add click handler to each button
       actionButtons.forEach(button => {
           button.addEventListener('click', handleVoteClick);
       });
       
       // Select all feedback dropdown submit buttons
       const submitFeedbackButtons = document.querySelectorAll('.submit-feedback-btn');
       
       // Add click handler to each submit button
       submitFeedbackButtons.forEach(button => {
           button.addEventListener('click', handleFeedbackSubmit);
       });
   }
   
   function handleVoteClick(event) {
       const button = event.currentTarget;
       const action = button.dataset.action;
       const jobCard = button.closest('.job-card');
       const jobId = jobCard.dataset.jobId;
       const matchId = jobCard.dataset.matchId;
       
       // Disable both buttons to prevent double-clicks
       jobCard.querySelectorAll('.interested-btn, .not-interested-btn')
           .forEach(btn => btn.disabled = true);
       
       // If action is "not-interested", show feedback dropdown
       if (action === 'not-interested') {
           const feedbackDropdown = jobCard.querySelector('.feedback-dropdown');
           feedbackDropdown.style.display = 'flex';
           return;
       }
       
       // For "interested" action, submit vote directly
       submitVote(jobCard, matchId, action, null);
   }
   
   function handleFeedbackSubmit(event) {
       const button = event.currentTarget;
       const jobCard = button.closest('.job-card');
       const matchId = jobCard.dataset.matchId;
       const feedbackSelect = jobCard.querySelector('.feedback-reason');
       const feedbackReason = feedbackSelect.value;
       
       // Validate feedback reason
       if (!feedbackReason) {
           // Highlight the dropdown to indicate required field
           feedbackSelect.classList.add('is-invalid');
           return;
       }
       
       // Submit vote with feedback
       submitVote(jobCard, matchId, 'not-interested', feedbackReason);
   }
   
   function submitVote(jobCard, matchId, action, feedbackReason) {
       // Prepare request data
       const data = {
           match_id: matchId,
           action: action,
           feedback_reason: feedbackReason
       };
       
       // Send AJAX request
       fetch('/api/job-feedback', {
           method: 'POST',
           headers: {
               'Content-Type': 'application/json',
               'X-CSRF-Token': getCSRFToken()
           },
           body: JSON.stringify(data)
       })
       .then(response => {
           if (!response.ok) {
               throw new Error('Network response was not ok');
           }
           return response.json();
       })
       .then(data => {
           // Handle successful response
           fadeOutCard(jobCard);
           loadNextCard();
       })
       .catch(error => {
           console.error('Error submitting vote:', error);
           showErrorMessage(jobCard, 'Failed to submit your feedback. Please try again.');
       });
   }
   
   function fadeOutCard(jobCard) {
       // Add fade-out class
       jobCard.classList.add('fade-out');
       
       // After animation completes, remove the card
       setTimeout(() => {
           const cardContainer = jobCard.parentElement;
           cardContainer.removeChild(jobCard);
       }, 500); // Match this to CSS animation duration
   }
   
   function loadNextCard() {
       // Load the next card to replace the one that was removed
       fetch('/api/next-job-match')
           .then(response => {
               if (!response.ok) {
                   throw new Error('Network response was not ok');
               }
               return response.json();
           })
           .then(data => {
               if (data.html) {
                   // Find an empty column or create one
                   const columns = document.querySelectorAll('.job-matches-container .col-md-4');
                   let targetColumn = null;
                   
                   for (const column of columns) {
                       if (column.children.length === 0) {
                           targetColumn = column;
                           break;
                       }
                   }
                   
                   if (!targetColumn) {
                       // Create a new column if needed
                       targetColumn = document.createElement('div');
                       targetColumn.classList.add('col-md-4');
                       document.querySelector('.job-matches-container .row').appendChild(targetColumn);
                   }
                   
                   // Insert the new card
                   targetColumn.innerHTML = data.html;
                   
                   // Initialize interaction for the new card
                   const newCard = targetColumn.querySelector('.job-card');
                   newCard.classList.add('fade-in');
                   newCard.querySelectorAll('.interested-btn, .not-interested-btn')
                       .forEach(btn => btn.addEventListener('click', handleVoteClick));
                   
                   newCard.querySelectorAll('.submit-feedback-btn')
                       .forEach(btn => btn.addEventListener('click', handleFeedbackSubmit));
               }
           })
           .catch(error => {
               console.error('Error loading next card:', error);
           });
   }
   
   function showErrorMessage(jobCard, message) {
       // Create error message element
       const errorDiv = document.createElement('div');
       errorDiv.classList.add('alert', 'alert-danger', 'mt-2');
       errorDiv.innerText = message;
       
       // Insert after the buttons
       jobCard.querySelector('.job-card-footer').appendChild(errorDiv);
       
       // Re-enable buttons
       jobCard.querySelectorAll('.interested-btn, .not-interested-btn')
           .forEach(btn => btn.disabled = false);
       
       // Remove error after a timeout
       setTimeout(() => {
           errorDiv.remove();
       }, 5000);
   }
   
   function getCSRFToken() {
       // Get CSRF token from meta tag
       return document.querySelector('meta[name="csrf-token"]').getAttribute('content');
   }
   ```

2. Create the server-side API handler for feedback in `/internal/handlers/feedback.go`:
   ```go
   package handlers

   import (
       "encoding/json"
       "net/http"
       "strconv"

       "github.com/talent-rodeo/internal/config"
   )

   // FeedbackRequest represents the JSON payload for job feedback
   type FeedbackRequest struct {
       MatchID        int    `json:"match_id"`
       Action         string `json:"action"`
       FeedbackReason string `json:"feedback_reason,omitempty"`
   }

   // HandleJobFeedback processes user feedback on job matches
   func (app *config.Application) HandleJobFeedback(w http.ResponseWriter, r *http.Request) {
       // Only allow POST requests
       if r.Method != http.MethodPost {
           app.clientError(w, http.StatusMethodNotAllowed)
           return
       }

       // Parse the request body
       var req FeedbackRequest
       err := json.NewDecoder(r.Body).Decode(&req)
       if err != nil {
           app.clientError(w, http.StatusBadRequest)
           return
       }

       // Validate the match ID
       if req.MatchID <= 0 {
           app.clientError(w, http.StatusBadRequest)
           return
       }

       // Get the current user ID
       user := app.authenticatedUser(r)
       if user == nil {
           app.clientError(w, http.StatusUnauthorized)
           return
       }

       // Verify the match belongs to the current user
       match, err := app.JobMatches.Get(req.MatchID)
       if err != nil || match.UserID != user.ID {
           app.clientError(w, http.StatusNotFound)
           return
       }

       // Process based on action
       var feedback string
       if req.Action == "interested" {
           feedback = "interested"
       } else if req.Action == "not-interested" {
           if req.FeedbackReason == "" {
               app.clientError(w, http.StatusBadRequest)
               return
           }
           feedback = "not_interested"
       } else {
           app.clientError(w, http.StatusBadRequest)
           return
       }

       // Update the job match with feedback
       err = app.JobMatches.UpdateFeedback(req.MatchID, feedback, req.FeedbackReason)
       if err != nil {
           app.serverError(w, err)
           return
       }

       // Return success response
       w.Header().Set("Content-Type", "application/json")
       json.NewEncoder(w).Encode(map[string]interface{}{
           "success": true,
           "message": "Feedback recorded successfully",
       })
   }

   // HandleNextJobMatch returns the next available job match
   func (app *config.Application) HandleNextJobMatch(w http.ResponseWriter, r *http.Request) {
       // Get the current user ID
       user := app.authenticatedUser(r)
       if user == nil {
           app.clientError(w, http.StatusUnauthorized)
           return
       }

       // Get viewed matches to exclude them
       viewedParam := r.URL.Query().Get("viewed")
       viewed := []int{}
       
       if viewedParam != "" {
           // Parse comma-separated match IDs
           for _, idStr := range strings.Split(viewedParam, ",") {
               if id, err := strconv.Atoi(idStr); err == nil {
                   viewed = append(viewed, id)
               }
           }
       }

       // Get the next unviewed match
       match, job, err := app.JobMatches.GetNextUnviewedMatch(user.ID, viewed)
       if err != nil {
           app.serverError(w, err)
           return
       }

       // If no match found, return empty response
       if match == nil {
           w.Header().Set("Content-Type", "application/json")
           json.NewEncoder(w).Encode(map[string]interface{}{
               "html": "",
               "no_more_matches": true,
           })
           return
       }

       // Render the job card template
       var buf bytes.Buffer
       err = app.renderPartial(&buf, r, "job_card", &templateData{
           Job:      job,
           JobMatch: match,
           Percentage: match.MatchPercentage,
       })

       if err != nil {
           app.serverError(w, err)
           return
       }

       // Return the rendered HTML
       w.Header().Set("Content-Type", "application/json")
       json.NewEncoder(w).Encode(map[string]interface{}{
           "html": buf.String(),
       })
   }

   // Helper function to render a partial template
   func (app *config.Application) renderPartial(w io.Writer, r *http.Request, name string, td *templateData) error {
       // Get the template from cache
       t, ok := app.templateCache[name]
       if !ok {
           return fmt.Errorf("the template %s does not exist", name)
       }

       // Prepare the data
       if td == nil {
           td = &templateData{}
       }
       td.CurrentYear = time.Now().Year()
       td.Flash = app.session.PopString(r, "flash")
       td.AuthenticatedUser = app.authenticatedUser(r)
       td.CSRFToken = nosurf.Token(r)

       // Render the template
       return t.Execute(w, td)
   }
   ```
