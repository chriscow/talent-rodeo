# T004.6.2: Card Display Implementation

## Task Description
Implement the card display area on the results dashboard that shows a limited number of job matches at a time with pagination controls. The interface should display up to three job cards simultaneously and provide navigation controls to browse through additional matches. This component is the central element of the results dashboard and should provide a smooth, intuitive browsing experience.

## Acceptance Criteria
1. The card display area shows up to three job cards at once
2. Cards are displayed in order of match percentage (highest first)
3. Pagination controls allow users to navigate through all available matches
4. Navigation is smooth with transition animations between pages
5. The component maintains state when a user provides feedback on a job
6. The layout is responsive and adjusts properly on mobile devices
7. If fewer than three job matches are available, the interface handles this gracefully
8. The display area communicates when all matches have been reviewed

3. Create JavaScript for pagination in `/static/js/job-pagination.js`:

   ```javascript
   document.addEventListener('DOMContentLoaded', function() {
       // Initialize pagination
       initPagination();
       
       // Handle card display transitions
       setupCardTransitions();
   });
   
   function initPagination() {
       const prevButton = document.getElementById('prev-page');
       const nextButton = document.getElementById('next-page');
       
       if (!prevButton || !nextButton) return;
       
       // Previous page handler
       prevButton.addEventListener('click', function(e) {
           if (this.classList.contains('disabled')) return;
           
           const currentPage = getCurrentPage();
           if (currentPage <= 1) return;
           
           navigateToPage(currentPage - 1);
       });
       
       // Next page handler
       nextButton.addEventListener('click', function(e) {
           if (this.classList.contains('disabled')) return;
           
           const currentPage = getCurrentPage();
           const totalPages = getTotalPages();
           if (currentPage >= totalPages) return;
           
           navigateToPage(currentPage + 1);
       });
   }
   
   function getCurrentPage() {
       const pageIndicator = document.querySelector('.page-indicator');
       if (!pageIndicator) return 1;
       
       const pageText = pageIndicator.textContent.trim();
       const match = pageText.match(/Page (\d+) of \d+/);
       return match ? parseInt(match[1], 10) : 1;
   }
   
   function getTotalPages() {
       const pageIndicator = document.querySelector('.page-indicator');
       if (!pageIndicator) return 1;
       
       const pageText = pageIndicator.textContent.trim();
       const match = pageText.match(/Page \d+ of (\d+)/);
       return match ? parseInt(match[1], 10) : 1;
   }
   
   function navigateToPage(page) {
       // Add loading state
       document.getElementById('job-cards-wrapper').classList.add('loading');
       
       // Make API request to get the page
       fetch(`/api/job-matches?page=${page}`, {
           method: 'GET',
           headers: {
               'Content-Type': 'application/json',
               'X-CSRF-Token': getCSRFToken()
           }
       })
       .then(response => {
           if (!response.ok) throw new Error('Network response was not ok');
           return response.json();
       })
       .then(data => {
           updateJobCards(data);
           updatePaginationControls(data.currentPage, data.totalPages);
           document.getElementById('job-cards-wrapper').classList.remove('loading');
       })
       .catch(error => {
           console.error('Error fetching job matches:', error);
           document.getElementById('job-cards-wrapper').classList.remove('loading');
           showErrorNotification('Failed to load job matches. Please try again.');
       });
   }
   
   function updateJobCards(data) {
       const jobCardsContainer = document.querySelector('.job-cards');
       if (!jobCardsContainer) return;
       
       // Fade out existing cards
       const existingCards = document.querySelectorAll('.job-card-wrapper');
       existingCards.forEach(card => {
           card.classList.add('fade-out');
       });
       
       // Clear and add new cards after animation
       setTimeout(() => {
           jobCardsContainer.innerHTML = '';
           
           if (data.jobs && data.jobs.length > 0) {
               data.jobs.forEach(job => {
                   const cardWrapper = document.createElement('div');
                   cardWrapper.className = 'job-card-wrapper fade-in';
                   cardWrapper.setAttribute('data-job-id', job.id);
                   cardWrapper.innerHTML = createJobCardHTML(job);
                   jobCardsContainer.appendChild(cardWrapper);
               });
               
               // Initialize card interactions for new cards
               if (typeof initJobCardInteractions === 'function') {
                   initJobCardInteractions();
               }
           } else {
               // No jobs to display
               jobCardsContainer.innerHTML = createNoJobsHTML();
           }
       }, 300);
   }
   
   function createJobCardHTML(job) {
       // This is a simplified version - the actual implementation would use the job card template
       return `
           <div class="job-card">
               <div class="job-card-header">
                   <div class="company-logo">
                       <img src="${job.companyLogo || '/static/img/placeholder-logo.png'}" alt="${job.companyName}">
                   </div>
                   <div class="job-title-container">
                       <h3 class="job-title">${job.title}</h3>
                       <div class="company-name">${job.companyName}</div>
                   </div>
                   <div class="match-indicator">
                       <div class="match-percentage">${job.matchPercentage}%</div>
                       <div class="match-bar">
                           <div class="match-fill" style="width: ${job.matchPercentage}%"></div>
                       </div>
                   </div>
               </div>
               <div class="job-card-body">
                   <div class="job-location"><i class="fas fa-map-marker-alt"></i> ${job.location}</div>
                   <div class="job-description">${job.description}</div>
                   <div class="match-explanation">${job.matchExplanation}</div>
               </div>
               <div class="job-card-footer">
                   <button class="btn btn-outline-danger not-interested" data-job-id="${job.id}">
                       Not Interested
                   </button>
                   <button class="btn btn-primary interested" data-job-id="${job.id}">
                       Interested
                   </button>
               </div>
           </div>
       `;
   }
   
   function createNoJobsHTML() {
       return `
           <div class="no-matches">
               <div class="no-matches-icon">
                   <i class="fas fa-search"></i>
               </div>
               <h3>No more job matches</h3>
               <p>You've reviewed all available job matches. Check back later for new opportunities.</p>
               <a href="/dashboard" class="btn btn-primary">Back to Dashboard</a>
           </div>
       `;
   }
   
   function updatePaginationControls(currentPage, totalPages) {
       const pageIndicator = document.querySelector('.page-indicator');
       const prevButton = document.getElementById('prev-page');
       const nextButton = document.getElementById('next-page');
       
       if (pageIndicator) {
           pageIndicator.textContent = `Page ${currentPage} of ${totalPages}`;
       }
       
       if (prevButton) {
           if (currentPage <= 1) {
               prevButton.classList.add('disabled');
           } else {
               prevButton.classList.remove('disabled');
           }
       }
       
       if (nextButton) {
           if (currentPage >= totalPages) {
               nextButton.classList.add('disabled');
           } else {
               nextButton.classList.remove('disabled');
           }
       }
   }
   
   function setupCardTransitions() {
       // Add entrance animation to initial cards
       document.querySelectorAll('.job-card-wrapper').forEach((card, index) => {
           // Stagger the animations
           setTimeout(() => {
               card.style.opacity = '1';
               card.style.transform = 'translateY(0)';
           }, index * 100);
       });
   }
   
   function getCSRFToken() {
       const tokenElement = document.querySelector('meta[name="csrf-token"]');
       return tokenElement ? tokenElement.getAttribute('content') : '';
   }
   
   function showErrorNotification(message) {
       // Implementation depends on notification system used
       console.error(message);
       // Example with alert for simplicity
       alert(message);
   }
   ```

4. Add the API endpoint for paginated job matches in `/internal/handlers/api.go`:

   ```go
   // Define response structure
   type JobMatchesResponse struct {
       Jobs        []ScoredJob `json:"jobs"`
       CurrentPage int         `json:"currentPage"`
       TotalPages  int         `json:"totalPages"`
   }
   
   // API handler for paginated job matches
   func (app *config.Application) APIJobMatches(w http.ResponseWriter, r *http.Request) {
       // Extract user ID from session
       userID := app.getUserIDFromSession(r)
       if userID == 0 {
           app.clientError(w, http.StatusUnauthorized)
           return
       }
       
       // Get page parameter
       page := 1
       pageParam := r.URL.Query().Get("page")
       if pageParam != "" {
           var err error
           page, err = strconv.Atoi(pageParam)
           if err != nil || page < 1 {
               page = 1
           }
       }
       
       // Page size is fixed at 3 for the MVP
       pageSize := 3
       
       // Get total count for pagination
       var totalJobs int
       err := app.DB.QueryRow(`
           SELECT COUNT(*) FROM job_matches 
           WHERE user_id = $1 AND status = 'new'
       `, userID).Scan(&totalJobs)
       
       if err != nil {
           app.serverError(w, err)
           return
       }
       
       // Calculate total pages
       totalPages := (totalJobs + pageSize - 1) / pageSize
       if totalPages == 0 {
           totalPages = 1
       }
       
       // Adjust page if out of bounds
       if page > totalPages {
           page = totalPages
       }
       
       // Calculate offset
       offset := (page - 1) * pageSize
       
       // Get paginated job matches
       rows, err := app.DB.Query(`
           SELECT jm.id, j.title, j.company_name, j.company_logo, j.location, 
                  j.description, jm.match_percentage, jm.match_explanation
           FROM job_matches jm
           JOIN jobs j ON jm.job_id = j.id
           WHERE jm.user_id = $1 AND jm.status = 'new'
           ORDER BY jm.match_percentage DESC
           LIMIT $2 OFFSET $3
       `, userID, pageSize, offset)
       
       if err != nil {
           app.serverError(w, err)
           return
       }
       defer rows.Close()
       
       // Build matches array
       var scoredJobs []ScoredJob
       for rows.Next() {
           var job ScoredJob
           err := rows.Scan(
               &job.ID, &job.Title, &job.CompanyName, &job.CompanyLogo,
               &job.Location, &job.Description, &job.MatchPercentage,
               &job.MatchExplanation,
           )
           
           if err != nil {
               app.serverError(w, err)
               return
           }
           
           // Truncate description for display
           if len(job.Description) > 200 {
               job.Description = job.Description[:197] + "..."
           }
           
           scoredJobs = append(scoredJobs, job)
       }
       
       // Check for any errors during row iteration
       if err = rows.Err(); err != nil {
           app.serverError(w, err)
           return
       }
       
       // Prepare response
       response := JobMatchesResponse{
           Jobs:        scoredJobs,
           CurrentPage: page,
           TotalPages:  totalPages,
       }
       
       // Send JSON response
       w.Header().Set("Content-Type", "application/json")
       json.NewEncoder(w).Encode(response)
   }
   ```
