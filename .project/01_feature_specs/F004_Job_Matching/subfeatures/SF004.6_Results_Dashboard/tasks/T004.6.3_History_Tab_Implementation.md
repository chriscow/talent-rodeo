# T004.6.3: History Tab Implementation

## Task Description
Implement a tabbed navigation system for the results dashboard that allows users to switch between current job matches and their match history. The history tab should display previously viewed job matches along with the feedback the user provided, allowing them to review past decisions and potentially revisit matches they were interested in.

## Acceptance Criteria
1. A tabbed navigation interface is implemented at the top of the results dashboard
2. Tabs include "Current Matches" and "History"
3. The history tab displays a list of previously viewed job matches
4. Each history entry shows:
   - Job title and company
   - Match percentage
   - The user's feedback (Interested/Not Interested)
   - Date when feedback was provided
5. History entries can be filtered by feedback type (All, Interested, Not Interested)
6. Users can revisit job details for history entries
7. The history tab is paginated if there are many entries
8. The active tab is visually distinguished
9. Tab state is preserved during the session

3. Create JavaScript for the history tab functionality in `/static/js/history-tab.js`:

   ```javascript
   document.addEventListener('DOMContentLoaded', function() {
       // Initialize tabs
       initTabs();
       
       // Initialize history filter controls
       initHistoryFilters();
   });
   
   function initTabs() {
       // Listen for tab changes to update URL
       const tabLinks = document.querySelectorAll('.nav-link[data-bs-toggle="tab"]');
       
       tabLinks.forEach(tabLink => {
           tabLink.addEventListener('shown.bs.tab', function(event) {
               // Get the active tab ID
               const activeTab = event.target.id;
               const tabName = activeTab.replace('-tab', '');
               
               // Update URL with the active tab
               const url = new URL(window.location);
               url.searchParams.set('tab', tabName);
               window.history.replaceState({}, '', url);
           });
       });
   }
   
   function initHistoryFilters() {
       // Handle filter button clicks
       const filterButtons = document.querySelectorAll('.history-controls .btn-group button');
       
       filterButtons.forEach(button => {
           button.addEventListener('click', function() {
               const filter = this.getAttribute('data-filter');
               
               // Update URL with the selected filter
               const url = new URL(window.location);
               url.searchParams.set('filter', filter);
               url.searchParams.set('page', '1');  // Reset to first page when filter changes
               
               // Navigate to the new URL
               window.location.href = url.toString();
           });
       });
   }
   ```

           // Create a slice for page numbers to show

           // Show at most 5 page numbers
           start := currentPage - 2
           if start < 1 {
               start = 1
           }
           
           end := start + 4
           if end > totalPages {
               end = totalPages
               // Adjust start to still show 5 pages if possible
               start = end - 4
               if start < 1 {
                   start = 1
               }
           }
           
           for i := start; i <= end; i++ {
               pages = append(pages, i)
           }
           
           return pages
       },
   }

   ```go
   // Update search handler to include tab navigation
   func (app *config.Application) HandleSearchResults(w http.ResponseWriter, r *http.Request) {
       // Extract user ID from session
       userID := app.getUserIDFromSession(r)
       if userID == 0 {
           http.Redirect(w, r, "/login", http.StatusSeeOther)
           return
       }
       
       // Determine active tab
       activeTab := r.URL.Query().Get("tab")
       if activeTab == "" {
           activeTab = "current" // Default tab
       }
       
       // Get search statistics
       searchStats, err := app.getSearchStats(userID)
       if err != nil {
           app.serverError(w, err)
           return
       }
       
       // Prepare template data
       data := &templateData{
           SearchStats: searchStats,
           ActiveTab:   activeTab,
       }
       
       // Handle current tab
       if activeTab == "current" {
           // Get page parameter
           page := 1
           pageParam := r.URL.Query().Get("page")
           if pageParam != "" {
               var err error
               page, err = strconv.Atoi(pageParam)
               if err != nil || page < 1 {
                   page = 1
               }
           }
           
           // Get current matches (code from previous task)
           currentJobs, totalPages, err := app.getCurrentJobMatches(userID, page)
           if err != nil {
               app.serverError(w, err)
               return
           }
           
           data.ScoredJobs = currentJobs
           data.CurrentPageJobs = currentJobs
           data.CurrentPage = page
           data.TotalPages = totalPages
       }
       
       // Handle history tab
       if activeTab == "history" {
           // Get filter parameter
           historyFilter := r.URL.Query().Get("filter")
           if historyFilter == "" {
               historyFilter = "all" // Default filter
           }
           
           // Get page parameter
           historyPage := 1
           pageParam := r.URL.Query().Get("page")
           if pageParam != "" {
               var err error
               historyPage, err = strconv.Atoi(pageParam)
               if err != nil || historyPage < 1 {
                   historyPage = 1
               }
           }
           
           // Get history matches
           historyJobs, historyTotalPages, err := app.getHistoryJobMatches(userID, historyFilter, historyPage)
           if err != nil {
               app.serverError(w, err)
               return
           }
           
           data.HistoryJobs = historyJobs
           data.HistoryFilter = historyFilter
           data.HistoryPage = historyPage
           data.HistoryTotalPages = historyTotalPages
       }
       
       // Render template
       app.render(w, r, "search_results.page.tmpl", data)
   }
   
   // Add method to get filtered history matches
   func (app *config.Application) getHistoryJobMatches(userID int, filter string, page int) ([]HistoryJob, int, error) {
       // Page size for history
       pageSize := 10
       
       // Base query
       baseQuery := `
           FROM job_matches jm
           JOIN jobs j ON jm.job_id = j.id
           WHERE jm.user_id = $1 AND jm.status != 'new'
       `
       
       // Apply filter
       var filterCondition string
       switch filter {
       case "interested":
           filterCondition = " AND jm.status = 'interested'"
       case "not-interested":
           filterCondition = " AND jm.status = 'not_interested'"
       default:
           filterCondition = "" // All history
       }
       
       // Get total count for pagination
       var totalJobs int
       countQuery := "SELECT COUNT(*) " + baseQuery + filterCondition
       err := app.DB.QueryRow(countQuery, userID).Scan(&totalJobs)
       if err != nil {
           return nil, 0, err
       }
       
       // Calculate total pages
       totalPages := (totalJobs + pageSize - 1) / pageSize
       if totalPages == 0 {
           totalPages = 1
       }
       
       // Adjust page if out of bounds
       if page > totalPages {
           page = totalPages
       }
       
       // Calculate offset
       offset := (page - 1) * pageSize
       
       // Get paginated and filtered job history
       query := `
           SELECT jm.id, j.title, j.company_name, jm.match_percentage, 
                  jm.status, jm.feedback_date
           ` + baseQuery + filterCondition + `
           ORDER BY jm.feedback_date DESC
           LIMIT $2 OFFSET $3
       `
       
       rows, err := app.DB.Query(query, userID, pageSize, offset)
       if err != nil {
           return nil, 0, err
       }
       defer rows.Close()
       
       // Build history array
       var historyJobs []HistoryJob
       for rows.Next() {
           var job HistoryJob
           var feedbackDate sql.NullTime
           
           err := rows.Scan(
               &job.ID, &job.Title, &job.CompanyName, 
               &job.MatchPercentage, &job.Status, &feedbackDate,
           )
           
           if err != nil {
               return nil, 0, err
           }
           
           // Handle nullable date
           if feedbackDate.Valid {
               job.FeedbackDate = feedbackDate.Time
           } else {
               job.FeedbackDate = time.Now() // Fallback
           }
           
           historyJobs = append(historyJobs, job)
       }
       
       // Check for any errors during row iteration
       if err = rows.Err(); err != nil {
           return nil, 0, err
       }
       
       return historyJobs, totalPages, nil
   }
   
   // Define HistoryJob struct
   type HistoryJob struct {
       ID              int
       Title           string
       CompanyName     string
       MatchPercentage float64
       Status          string
       FeedbackDate    time.Time
   }
   ```

## Technical Notes
- The tab state is maintained through URL parameters
- History filters allow users to quickly find specific feedback types
- Table design is responsive but may require horizontal scrolling on small devices
- Pagination is server-side with page range logic to avoid showing too many page links
- The implementation separates concerns between the two tabs while sharing the status bar
- Color coding is used for match percentages and feedback types for quick visual scanning
- URLs are constructed to preserve filter and tab state during pagination
