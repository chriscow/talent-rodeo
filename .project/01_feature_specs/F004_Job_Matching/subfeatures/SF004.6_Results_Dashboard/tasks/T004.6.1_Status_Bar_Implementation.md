# T004.6.1: Status Bar Implementation

## Task Description
Implement a status bar component for the job matching results dashboard that displays search progress metrics and key information about the match process. The status bar should remain visible at the top of the page during scrolling and provide users with contextual information about their job search process.

## Acceptance Criteria
1. A non-scrolling status bar is implemented at the top of the results page
2. The status bar displays:
   - Number of jobs searched
   - Number of matches found above threshold
   - Next scheduled search time
   - Subscription tier information
3. The status bar has a clean, informative design that aligns with the application style
4. The component is responsive and works well on both desktop and mobile devices
5. Content updates dynamically when new search results are available
6. The status bar remains visible during scrolling (sticky positioning)

3. Add helper functions for formatting and tier information in `/cmd/web/templates.go`:
   ```go
   // Add to template functions
   functions := template.FuncMap{
       // Existing functions
       
       "formatDateTime": func(t time.Time) string {
           // If less than 24 hours, show relative time
           if time.Since(t) < -time.Hour*24 {
               return t.Format("Jan 2, 3:04 PM")
           }
           
           // If future time today, show "Today at X:XX"
           if t.Year() == time.Now().Year() && t.YearDay() == time.Now().YearDay() {
               return "Today at " + t.Format("3:04 PM")
           }
           
           // If tomorrow, show "Tomorrow at X:XX"
           tomorrow := time.Now().AddDate(0, 0, 1)
           if t.Year() == tomorrow.Year() && t.YearDay() == tomorrow.YearDay() {
               return "Tomorrow at " + t.Format("3:04 PM")
           }
           
           // Otherwise, show full date/time
           return t.Format("Jan 2 at 3:04 PM")
       },
       
       "subscriptionInfo": func(tier string) string {
           switch strings.ToLower(tier) {
           case "free":
               return "Weekly searches"
           case "basic":
               return "Daily searches"
           case "premium":
               return "Continuous updates"
           default:
               return ""
           }
       },
   }
   ```

4. Update the search handler to include search statistics in `/internal/handlers/search.go`:
   ```go
   // Add a SearchStats struct
   type SearchStats struct {
       JobsSearched    int
       MatchesFound    int
       NextSearchTime  time.Time
   }
   
   // Update search handler to include stats
   func (app *config.Application) HandleSearchResults(w http.ResponseWriter, r *http.Request) {
       // Existing code to get user and matches
       
       // Get search statistics
       searchStats, err := app.getSearchStats(userID)
       if err != nil {
           app.serverError(w, err)
           return
       }
       
       // Render template with stats
       app.render(w, r, "search_results.page.tmpl", &templateData{
           ScoredJobs:  scoredJobs,
           SearchStats: searchStats,
       })
   }
   
   // Add method to get search statistics
   func (app *config.Application) getSearchStats(userID int) (*SearchStats, error) {
       // Get job search count from database
       var jobsSearched int
       err := app.DB.QueryRow(`
           SELECT COUNT(*) FROM job_searches 
           WHERE user_id = $1
       `, userID).Scan(&jobsSearched)
       
       if err != nil {
           return nil, err
       }
       
       // Get matches count from database
       var matchesFound int
       err = app.DB.QueryRow(`
           SELECT COUNT(*) FROM job_matches 
           WHERE user_id = $1 AND match_percentage >= $2
       `, userID, app.MatchThreshold).Scan(&matchesFound)
       
       if err != nil {
           return nil, err
       }
       
       // Get user's subscription tier
       var tier string
       err = app.DB.QueryRow(`
           SELECT subscription_tier FROM users
           WHERE id = $1
       `, userID).Scan(&tier)
       
       if err != nil {
           return nil, err
       }
       
       // Calculate next search time based on tier
       var nextSearchTime time.Time
       now := time.Now()
       
       switch strings.ToLower(tier) {
       case "free":
           // Weekly: Next Monday
           daysUntilMonday := (8 - int(now.Weekday())) % 7
           if daysUntilMonday == 0 {
               daysUntilMonday = 7
           }
           nextSearchTime = time.Date(now.Year(), now.Month(), now.Day(), 9, 0, 0, 0, now.Location()).
               AddDate(0, 0, daysUntilMonday)
       case "basic":
           // Daily: Tomorrow morning
           nextSearchTime = time.Date(now.Year(), now.Month(), now.Day(), 9, 0, 0, 0, now.Location()).
               AddDate(0, 0, 1)
       case "premium":
           // Continuous: In 4 hours
           nextSearchTime = now.Add(4 * time.Hour)
       default:
           // Default to weekly
           daysUntilMonday := (8 - int(now.Weekday())) % 7
           if daysUntilMonday == 0 {
               daysUntilMonday = 7
           }
           nextSearchTime = time.Date(now.Year(), now.Month(), now.Day(), 9, 0, 0, 0, now.Location()).
               AddDate(0, 0, daysUntilMonday)
       }
       
       return &SearchStats{
           JobsSearched:   jobsSearched,
           MatchesFound:   matchesFound,
           NextSearchTime: nextSearchTime,
       }, nil
   }
   ```
