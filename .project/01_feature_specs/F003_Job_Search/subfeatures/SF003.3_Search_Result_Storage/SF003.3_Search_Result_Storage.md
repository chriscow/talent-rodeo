# SF003.3: Search Result Storage

## Overview
The Search Result Storage subfeature enables Talent Rodeo to efficiently store, retrieve, and manage job search results. This creates a foundation for the matching process, supports user dashboard views, and enables historical tracking of job search activity.

## Implementation Status
**Partially Implemented**

## Detailed Requirements

### Data Storage
1. Store complete job listings in a structured database format
2. Maintain associations between search queries and results
3. Support efficient retrieval of job listings by various attributes
4. Implement versioning for job listings that change over time
5. Store metadata about search execution and performance

### Data Organization
1. Organize search results by search query
2. Track result order and relevance within a search
3. Support paging through large result sets
4. Implement data deduplication for jobs appearing in multiple searches
5. Preserve original source information for each job listing

### Result Lifecycle Management
1. Implement timestamp tracking for result freshness
2. Support automatic expiration of outdated search results
3. Implement purging of old or invalid job listings
4. Track job posting status (active, filled, expired)
5. Support manual refresh of specific search results

### Indexing and Retrieval
1. Implement efficient indexing for job listings
2. Support fast retrieval by common search attributes
3. Enable full-text search within job descriptions
4. Support complex queries across multiple attributes
5. Optimize database schema for common access patterns

## User Experience
- Users should be able to view past search results
- System should track which jobs the user has viewed or interacted with
- Search history should be accessible and re-executable
- Job status changes should be reflected in user views
- Search results should load quickly regardless of volume

## Database Schema References
```sql
-- job_listings table for storing detailed job information
CREATE TABLE IF NOT EXISTS job_listings (
    id SERIAL PRIMARY KEY,
    external_id TEXT NOT NULL UNIQUE,
    title TEXT NOT NULL,
    company TEXT NOT NULL,
    location TEXT,
    description TEXT,
    url TEXT NOT NULL,
    posted_date TIMESTAMP,
    status VARCHAR(20) NOT NULL DEFAULT 'active',
    source TEXT NOT NULL DEFAULT 'linkedin',
    metadata JSONB,
    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP
);

-- job_searches table for tracking search queries
CREATE TABLE IF NOT EXISTS job_searches (
    id SERIAL PRIMARY KEY,
    candidate_id INTEGER NOT NULL REFERENCES candidates(id),
    query_hash TEXT NOT NULL,
    query_text TEXT NOT NULL,
    search_params JSONB NOT NULL,
    result_count INTEGER NOT NULL DEFAULT 0,
    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    expires_at TIMESTAMP NOT NULL,
    UNIQUE (candidate_id, query_hash)
);

-- search_results table linking searches to job listings
CREATE TABLE IF NOT EXISTS search_results (
    id SERIAL PRIMARY KEY,
    search_id INTEGER NOT NULL REFERENCES job_searches(id),
    job_id INTEGER NOT NULL REFERENCES job_listings(id),
    result_order INTEGER NOT NULL,
    relevance_score FLOAT,
    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    UNIQUE (search_id, job_id)
);

-- candidate_job_interactions for tracking user interaction with jobs
CREATE TABLE IF NOT EXISTS candidate_job_interactions (
    id SERIAL PRIMARY KEY,
    candidate_id INTEGER NOT NULL REFERENCES candidates(id),
    job_id INTEGER NOT NULL REFERENCES job_listings(id),
    interaction_type VARCHAR(50) NOT NULL, -- 'view', 'save', 'apply', etc.
    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    UNIQUE (candidate_id, job_id, interaction_type)
);
```

## Usage Examples

### Storing Search Results
```go
// Sample code for storing search results
func StoreJobSearchResults(ctx context.Context, search *JobSearch, jobs []*JobListing) error {
    tx, err := db.BeginTx(ctx, nil)
    if err != nil {
        return fmt.Errorf("failed to begin transaction: %w", err)
    }
    defer tx.Rollback()
    
    // Insert or update the search record
    var searchID int
    err = tx.QueryRowContext(
        ctx,
        "INSERT INTO job_searches (candidate_id, query_hash, query_text, search_params, result_count, expires_at) VALUES ($1, $2, $3, $4, $5, $6) "+
        "ON CONFLICT (candidate_id, query_hash) DO UPDATE SET result_count = $5, expires_at = $6, updated_at = CURRENT_TIMESTAMP "+
        "RETURNING id",
        search.CandidateID, search.QueryHash, search.QueryText, search.Params, len(jobs), search.ExpiresAt,
    ).Scan(&searchID)
    if err != nil {
        return fmt.Errorf("failed to store search: %w", err)
    }
    
    // Insert or update job listings and create search result records
    for i, job := range jobs {
        var jobID int
        err = tx.QueryRowContext(
            ctx,
            "INSERT INTO job_listings (external_id, title, company, location, description, url, posted_date, source, metadata) "+
            "VALUES ($1, $2, $3, $4, $5, $6, $7, $8, $9) "+
            "ON CONFLICT (external_id) DO UPDATE SET "+
            "title = $2, company = $3, location = $4, description = $5, url = $6, updated_at = CURRENT_TIMESTAMP "+
            "RETURNING id",
            job.ExternalID, job.Title, job.Company, job.Location, job.Description, job.URL, job.PostedDate, job.Source, job.Metadata,
        ).Scan(&jobID)
        if err != nil {
            return fmt.Errorf("failed to store job listing: %w", err)
        }
        
        // Create search result record
        _, err = tx.ExecContext(
            ctx,
            "INSERT INTO search_results (search_id, job_id, result_order, relevance_score) "+
            "VALUES ($1, $2, $3, $4) "+
            "ON CONFLICT (search_id, job_id) DO UPDATE SET result_order = $3, relevance_score = $4",
            searchID, jobID, i, job.RelevanceScore,
        )
        if err != nil {
            return fmt.Errorf("failed to create search result record: %w", err)
        }
    }
    
    return tx.Commit()
}
```

### Retrieving Search Results
```go
// Sample code for retrieving search results
func GetJobSearchResults(ctx context.Context, candidateID int, queryHash string, page, pageSize int) (*SearchResultPage, error) {
    offset := (page - 1) * pageSize
    
    // Get search information
    var search JobSearch
    err := db.QueryRowContext(
        ctx,
        "SELECT id, query_text, search_params, result_count, created_at, expires_at FROM job_searches "+
        "WHERE candidate_id = $1 AND query_hash = $2",
        candidateID, queryHash,
    ).Scan(&search.ID, &search.QueryText, &search.Params, &search.ResultCount, &search.CreatedAt, &search.ExpiresAt)
    if err != nil {
        return nil, fmt.Errorf("failed to retrieve search: %w", err)
    }
    
    // Get paginated job results
    rows, err := db.QueryContext(
        ctx,
        "SELECT j.id, j.external_id, j.title, j.company, j.location, j.description, j.url, j.posted_date, j.status, j.source, "+
        "sr.relevance_score, EXISTS(SELECT 1 FROM candidate_job_interactions WHERE candidate_id = $1 AND job_id = j.id) AS interacted "+
        "FROM job_listings j "+
        "JOIN search_results sr ON sr.job_id = j.id "+
        "WHERE sr.search_id = $2 "+
        "ORDER BY sr.result_order "+
        "LIMIT $3 OFFSET $4",
        candidateID, search.ID, pageSize, offset,
    )
    if err != nil {
        return nil, fmt.Errorf("failed to retrieve search results: %w", err)
    }
    defer rows.Close()
    
    // Process results
    var jobs []*JobListing
    for rows.Next() {
        var job JobListing
        var interacted bool
        err := rows.Scan(
            &job.ID, &job.ExternalID, &job.Title, &job.Company, &job.Location, &job.Description, &job.URL, 
            &job.PostedDate, &job.Status, &job.Source, &job.RelevanceScore, &interacted,
        )
        if err != nil {
            return nil, fmt.Errorf("failed to scan job result: %w", err)
        }
        job.Interacted = interacted
        jobs = append(jobs, &job)
    }
    
    return &SearchResultPage{
        Search:    search,
        Jobs:      jobs,
        Page:      page,
        PageSize:  pageSize,
        TotalJobs: search.ResultCount,
        TotalPages: (search.ResultCount + pageSize - 1) / pageSize,
    }, nil
}
```
