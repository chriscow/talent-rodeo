# SF002.2.5: Time Management Strategies

## Overview
This specification details the time management strategies used by the TimeManagementHandler (defined in SF002.2.2) to ensure assessment calls remain within the 15-20 minute target duration. It provides comprehensive guidance on time tracking, pacing, and adaptive time allocation strategies.

This implementation builds on the event handler pattern established for the OpenAI realtime API:

```go
// Time Management Handler implementation
type TimeManagementHandler struct {
    startTime       time.Time
    maxDuration     time.Duration
    phaseStartTimes map[string]time.Time
    phaseAllocations map[string]time.Duration
    phaseManager    *PhaseManager
    db              *sql.DB
    conversationID  string
    lastMetricLog   time.Time
    mu              sync.RWMutex
}

func NewTimeManagementHandler(db *sql.DB, conversationID string, maxDuration time.Duration, phaseConfig *PhaseConfig) *TimeManagementHandler {
    // Calculate target durations for each phase based on percentages
    phaseAllocations := make(map[string]time.Duration)
    for phase, config := range phaseConfig.Phases {
        allocation := config.TimeAllocationPercentage
        if allocation == 0 {
            allocation = 0.2 // Default 20% if not specified
        }
        phaseDuration := time.Duration(float64(maxDuration) * allocation)
        phaseAllocations[phase] = phaseDuration
    }
    
    return &TimeManagementHandler{
        startTime:       time.Now(),
        maxDuration:     maxDuration,
        phaseStartTimes: make(map[string]time.Time),
        phaseAllocations: phaseAllocations,
        db:              db,
        conversationID:  conversationID,
        lastMetricLog:   time.Now(),
    }
}

func (h *TimeManagementHandler) SetPhaseManager(pm *PhaseManager) {
    h.phaseManager = pm
}

// Implement EventHandler interface
func (h *TimeManagementHandler) HandleEvent(event Event) error {
    // Process realtime API events
    switch e := event.(type) {
    case *ConversationStarted:
        return h.handleConversationStart(e)
    case *PhaseChanged:
        return h.handlePhaseChange(e)
    case *ResponseAudioTranscriptDelta:
        return h.handleTranscriptDelta(e)
    case *ResponseAudioTranscriptDone:
        return h.handleTranscriptDone(e)
    case *ResponseMessagePartDone:
        return h.handleMessagePartDone(e)
    case *TimerTick:
        return h.handleTimerTick(e)
    }
    return nil
}

func (h *TimeManagementHandler) handleConversationStart(e *ConversationStarted) error {
    h.mu.Lock()
    h.startTime = time.Now()
    h.mu.Unlock()
    return nil
}

func (h *TimeManagementHandler) handlePhaseChange(e *PhaseChanged) error {
    h.mu.Lock()
    h.phaseStartTimes[e.Phase] = time.Now()
    h.mu.Unlock()
    return nil
}

func (h *TimeManagementHandler) handleTimerTick(e *TimerTick) error {
    // This method is called periodically (e.g., every 30 seconds)
    // to evaluate timing and potentially force phase transitions
    
    // Calculate timing metrics
    metrics := h.calculateTimingMetrics()
    
    // Log metrics every minute
    if time.Since(h.lastMetricLog) > time.Minute {
        h.logTimingMetrics(metrics)
        h.lastMetricLog = time.Now()
    }
    
    // Check if we need to force a phase transition due to time constraints
    if h.shouldForceTransition(metrics) {
        h.forcePhaseTransition(metrics)
    }
    
    return nil
}

func (h *TimeManagementHandler) handleTranscriptDone(e *ResponseAudioTranscriptDone) error {
    // Process complete transcript
    // This is a good time to check overall pace and timing
    
    metrics := h.calculateTimingMetrics()
    
    // Update conversation context with timing guidance for next response
    guidanceTemplate := h.selectPacingGuidance(metrics)
    log.Printf("Time management guidance: %s", guidanceTemplate)
    
    return nil
}

func (h *TimeManagementHandler) handleMessagePartDone(e *ResponseMessagePartDone) error {
    // Process LLM response completion
    // This is a good point to check if time-based guidance was followed
    return nil
}

func (h *TimeManagementHandler) calculateTimingMetrics() *TimingMetrics {
    h.mu.RLock()
    defer h.mu.RUnlock()
    
    currentPhase := ""
    if h.phaseManager != nil {
        currentPhase = h.phaseManager.GetCurrentPhase()
    }
    
    elapsedTotal := time.Since(h.startTime)
    remainingTotal := h.maxDuration - elapsedTotal
    
    // Default values for phase metrics
    var elapsedPhase time.Duration
    var remainingPhase time.Duration
    
    if currentPhase != "" {
        phaseStart, exists := h.phaseStartTimes[currentPhase]
        if exists {
            elapsedPhase = time.Since(phaseStart)
            allocation := h.phaseAllocations[currentPhase]
            remainingPhase = allocation - elapsedPhase
        }
    }
    
    return &TimingMetrics{
        PhaseStartTime:  h.phaseStartTimes[currentPhase],
        PhaseElapsed:    elapsedPhase,
        PhaseRemaining:  remainingPhase,
        TotalElapsed:    elapsedTotal,
        TotalRemaining:  remainingTotal,
        PaceMetric:      calculatePaceMetric(elapsedTotal, h.maxDuration),
    }
}

func (h *TimeManagementHandler) selectPacingGuidance(metrics *TimingMetrics) string {
    // Calculate percentages for readable context
    percentComplete := float64(metrics.TotalElapsed) / float64(h.maxDuration) * 100
    
    var percentPhaseComplete float64
    if metrics.PhaseElapsed > 0 && h.phaseAllocations[h.phaseManager.GetCurrentPhase()] > 0 {
        percentPhaseComplete = float64(metrics.PhaseElapsed) / 
            float64(h.phaseAllocations[h.phaseManager.GetCurrentPhase()]) * 100
    }
    
    // Time status string
    timeInfo := fmt.Sprintf(
        "TIME_INFO: Call duration: %s elapsed (%.0f%%), %s remaining. Phase '%s': %s elapsed (%.0f%%), %s remaining.",
        formatDuration(metrics.TotalElapsed),
        percentComplete,
        formatDuration(metrics.TotalRemaining),
        h.phaseManager.GetCurrentPhase(),
        formatDuration(metrics.PhaseElapsed),
        percentPhaseComplete,
        formatDuration(metrics.PhaseRemaining),
    )
    
    // Select appropriate guidance
    var pacingGuidance string
    
    if percentComplete > 90 && h.phaseManager.GetCurrentPhase() != "conclusion" {
        pacingGuidance = pacingTemplates["critical_time_pressure"]
    } else if percentComplete > 75 && percentPhaseComplete > 100 {
        pacingGuidance = pacingTemplates["significant_time_pressure"]
    } else if percentComplete > 60 && percentPhaseComplete > 80 {
        pacingGuidance = pacingTemplates["moderate_time_pressure"]
    } else if percentComplete < 30 && percentPhaseComplete < 20 {
        pacingGuidance = pacingTemplates["ahead_of_schedule"]
    } else {
        pacingGuidance = pacingTemplates["on_schedule"]
    }
    
    return timeInfo + "\n" + pacingGuidance
}

func (h *TimeManagementHandler) shouldForceTransition(metrics *TimingMetrics) bool {
    currentPhase := h.phaseManager.GetCurrentPhase()
    
    // Never force transition from conclusion phase
    if currentPhase == "conclusion" {
        return false
    }
    
    // Critical time pressure - force to conclusion if near max duration
    if metrics.TotalRemaining < 2*time.Minute {
        return true
    }
    
    // Phase taking too long - force transition if significantly over allocation
    phaseAllocation := h.phaseAllocations[currentPhase]
    if metrics.PhaseElapsed > phaseAllocation*2 {
        return true
    }
    
    return false
}

func (h *TimeManagementHandler) forcePhaseTransition(metrics *TimingMetrics) {
    currentPhase := h.phaseManager.GetCurrentPhase()
    
    // If near the end of total time, force to conclusion
    if metrics.TotalRemaining < 2*time.Minute && currentPhase != "conclusion" {
        h.phaseManager.TransitionToPhase("conclusion")
        log.Printf("FORCED TRANSITION: Critical time pressure. Skipping to conclusion phase.")
        return
    }
    
    // Otherwise just go to next phase
    h.phaseManager.TransitionToNextPhase()
    log.Printf("FORCED TRANSITION: Phase %s exceeded time allocation. Moving to next phase.", currentPhase)
}

func (h *TimeManagementHandler) logTimingMetrics(metrics *TimingMetrics) {
    currentPhase := ""
    if h.phaseManager != nil {
        currentPhase = h.phaseManager.GetCurrentPhase()
    }
    
    if currentPhase == "" {
        return
    }
    
    _, err := h.db.Exec(`
        INSERT INTO conversation_timing_metrics (
            conversation_id, phase_id, start_time, duration_seconds, expected_duration_seconds, time_efficiency
        ) VALUES (?, ?, ?, ?, ?, ?)`,
        h.conversationID,
        currentPhase,
        metrics.PhaseStartTime,
        int(metrics.PhaseElapsed.Seconds()),
        int(h.phaseAllocations[currentPhase].Seconds()),
        metrics.PaceMetric,
    )
    
    if err != nil {
        log.Printf("Failed to log timing metrics: %v", err)
    }
}

// Helper function to calculate pace metric
func calculatePaceMetric(elapsed, total time.Duration) float64 {
    if total == 0 {
        return 1.0
    }
    expectedCompletion := float64(elapsed) / float64(total)
    return expectedCompletion
}

// Helper function to format durations in a user-friendly way
func formatDuration(d time.Duration) string {
    minutes := int(d.Minutes())
    seconds := int(d.Seconds()) % 60
    return fmt.Sprintf("%dm%02ds", minutes, seconds)
}
```

## System Overview

The Time Management component ensures assessment calls adhere to the required 15-20 minute duration while maintaining natural conversation flow. It acts as the "timekeeper" for the conversation, tracking progress and making strategic adjustments to ensure all critical information is collected within the time constraints.

Key aspects of this system include:

1. **Multi-Level Time Tracking**: Monitors time at the overall conversation, phase, and topic levels
2. **Adaptive Pacing**: Adjusts the conversation pace based on time progress and priorities
3. **Phase Transition Management**: Ensures appropriate time allocation across conversation phases
4. **Priority-Based Time Allocation**: Focuses remaining time on gathering the most critical information
5. **Graceful Conclusion**: Ensures conversations end appropriately within time constraints
6. **Time-Aware Communication**: Modifies system responses based on time conditions
