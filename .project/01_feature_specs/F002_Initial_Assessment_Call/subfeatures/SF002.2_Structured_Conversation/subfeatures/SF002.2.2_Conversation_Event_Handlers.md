# SF002.2.2: Conversation Event Handlers

## Overview
The Conversation Event Handlers provide essential cross-cutting functionality for managing conversation flow, logging, persistence, and monitoring. These handlers integrate with the OpenAI realtime API to process events, maintain conversation state, and provide critical services like logging, persistence, summarization, and error recovery.

## Detailed Requirements

### Core Event Handler Infrastructure
1. Implement a common event handler interface:
   ```go
   type EventHandler interface {
       // HandleEvent processes a single realtime API event
       HandleEvent(ctx context.Context, event openairt.ServerEvent) error
       
       // Initialize sets up the handler with necessary dependencies
       Initialize(ctx context.Context) error
       
       // Cleanup performs any necessary cleanup when conversation ends
       Cleanup(ctx context.Context) error
       
       // HandlePhaseTransition is called when phases change
       HandlePhaseTransition(ctx context.Context, from, to string) error
   }
   ```

2. Create a registry system for event handlers:
   ```go
   type HandlerRegistry struct {
       handlers    []EventHandler
       logger      *slog.Logger
       mu          sync.RWMutex
   }

   func (r *HandlerRegistry) Register(handler EventHandler) {
       r.mu.Lock()
       defer r.mu.Unlock()
       r.handlers = append(r.handlers, handler)
   }

   func (r *HandlerRegistry) ProcessEvent(ctx context.Context, event openairt.ServerEvent) error {
       r.mu.RLock()
       defer r.mu.RUnlock()
       
       for _, handler := range r.handlers {
           if err := handler.HandleEvent(ctx, event); err != nil {
               r.logger.Error("handler error", 
                   "handler", fmt.Sprintf("%T", handler),
                   "error", err)
           }
       }
       return nil
   }
   ```

### Standard Event Handlers

1. **Conversation Logger**:
   ```go
   type ConversationLogger struct {
       db     *sql.DB
       logger *slog.Logger
   }

   func (l *ConversationLogger) HandleEvent(ctx context.Context, event openairt.ServerEvent) error {
       switch event.ServerEventType() {
       case openairt.ServerEventTypeResponseAudioTranscriptDelta:
           delta := event.(openairt.ResponseAudioTranscriptDeltaEvent)
           l.logger.Info("transcript delta", 
               "content", delta.Delta,
               "role", "candidate")
           
       case openairt.ServerEventTypeResponseAudioTranscriptDone:
           l.logger.Info("transcript complete", 
               "role", "candidate")
       }
       return nil
   }
   ```

2. **Persistence Handler**:
   ```go
   type PersistenceHandler struct {
       db              *sql.DB
       conversationID  string
       transcriptBuf   strings.Builder
       transcriptMu    sync.Mutex
   }

   func (h *PersistenceHandler) HandleEvent(ctx context.Context, event openairt.ServerEvent) error {
       switch event.ServerEventType() {
       case openairt.ServerEventTypeResponseAudioTranscriptDelta:
           delta := event.(openairt.ResponseAudioTranscriptDeltaEvent)
           h.transcriptMu.Lock()
           h.transcriptBuf.WriteString(delta.Delta)
           h.transcriptMu.Unlock()
           
       case openairt.ServerEventTypeResponseAudioTranscriptDone:
           h.transcriptMu.Lock()
           transcript := h.transcriptBuf.String()
           h.transcriptBuf.Reset()
           h.transcriptMu.Unlock()
           
           return h.persistTranscript(ctx, transcript)
       }
       return nil
   }
   ```

3. **Summarization Handler**:
   ```go
   type SummarizationHandler struct {
       llmClient      LLMClient
       db            *sql.DB
       conversationID string
       summaryBuf    strings.Builder
       summaryMu     sync.Mutex
   }

   func (h *SummarizationHandler) HandleEvent(ctx context.Context, event openairt.ServerEvent) error {
       switch event.ServerEventType() {
       case openairt.ServerEventTypeResponseAudioTranscriptDone:
           // Get complete transcript
           transcript := h.getCompleteTranscript()
           
           // Generate summary asynchronously
           go h.generateAndStoreSummary(ctx, transcript)
       }
       return nil
   }
   ```

4. **Error Recovery Handler**:
   ```go
   type ErrorRecoveryHandler struct {
       phaseManager   *PhaseManager
       errorCount     int
       errorThreshold int
       lastActivity   time.Time
       mu            sync.Mutex
   }

   func (h *ErrorRecoveryHandler) HandleEvent(ctx context.Context, event openairt.ServerEvent) error {
       h.mu.Lock()
       defer h.mu.Unlock()
       
       // Reset error count on successful interaction
       switch event.ServerEventType() {
       case openairt.ServerEventTypeResponseAudioTranscriptDone:
           h.errorCount = 0
           h.lastActivity = time.Now()
           
       case openairt.ServerEventTypeError:
           h.errorCount++
           if h.errorCount >= h.errorThreshold {
               return h.initiateErrorRecovery(ctx)
           }
       }
       return nil
   }
   ```

5. **Time Management Handler**:
   ```go
   type TimeManagementHandler struct {
       phaseManager    *PhaseManager
       startTime       time.Time
       phaseDurations  map[string]time.Duration
       maxDuration     time.Duration
   }

   func (h *TimeManagementHandler) HandleEvent(ctx context.Context, event openairt.ServerEvent) error {
       // Check overall duration
       if time.Since(h.startTime) > h.maxDuration {
           return h.initiateConversationEnd(ctx)
       }
       
       // Check phase duration
       currentPhase := h.phaseManager.CurrentPhase()
       if h.isPhaseTimeExceeded(currentPhase) {
           return h.phaseManager.TransitionToNextPhase(ctx)
       }
       
       return nil
   }
   ```

### Custom Handler Example

```go
type NonResponseDetector struct {
    phaseManager    *PhaseManager
    silenceTimeout  time.Duration
    lastActivity    time.Time
    mu             sync.Mutex
}

func NewNonResponseDetector(timeout time.Duration) *NonResponseDetector {
    return &NonResponseDetector{
        silenceTimeout: timeout,
        lastActivity:  time.Now(),
    }
}

func (d *NonResponseDetector) HandleEvent(ctx context.Context, event openairt.ServerEvent) error {
    d.mu.Lock()
    defer d.mu.Unlock()

    switch event.ServerEventType() {
    case openairt.ServerEventTypeResponseAudioTranscriptDelta:
        d.lastActivity = time.Now()
        
    case openairt.ServerEventTypeResponseAudioTranscriptDone:
        if time.Since(d.lastActivity) > d.silenceTimeout {
            return d.handleNonResponse(ctx)
        }
    }
    return nil
}

func (d *NonResponseDetector) handleNonResponse(ctx context.Context) error {
    // Implement non-response recovery strategy
    return d.phaseManager.RequestTransition(ctx, "error_recovery")
}
```

### Integration with ConversationHandler

```go
type ConversationHandler struct {
    client         *openairt.Client
    conn           *openairt.Conn
    registry       *HandlerRegistry
    phaseManager   *PhaseManager
    logger         *slog.Logger
}

func NewConversationHandler(config *Config) (*ConversationHandler, error) {
    handler := &ConversationHandler{
        client:   openairt.NewClient(config.APIKey),
        registry: NewHandlerRegistry(),
        logger:   config.Logger,
    }
    
    // Register standard handlers
    handler.registry.Register(NewConversationLogger(config.DB))
    handler.registry.Register(NewPersistenceHandler(config.DB))
    handler.registry.Register(NewSummarizationHandler(config.LLMClient))
    handler.registry.Register(NewErrorRecoveryHandler(config.ErrorThreshold))
    handler.registry.Register(NewTimeManagementHandler(config.MaxDuration))
    handler.registry.Register(NewNonResponseDetector(5 * time.Second))
    
    return handler, nil
}

func (h *ConversationHandler) Start(ctx context.Context) error {
    // Initialize all handlers
    for _, handler := range h.registry.handlers {
        if err := handler.Initialize(ctx); err != nil {
            return fmt.Errorf("handler init error: %w", err)
        }
    }
    
    // Start processing events
    for {
        event, err := h.conn.ReadMessage(ctx)
        if err != nil {
            return fmt.Errorf("read error: %w", err)
        }
        
        // Process event through all handlers
        if err := h.registry.ProcessEvent(ctx, event); err != nil {
            h.logger.Error("event processing error", "error", err)
        }
    }
}
```

## Database Schema

```sql
-- Handler configurations
CREATE TABLE IF NOT EXISTS handler_configurations (
    id TEXT PRIMARY KEY,
    handler_type TEXT NOT NULL,
    config_json TEXT NOT NULL,
    is_active BOOLEAN DEFAULT true,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP
);

-- Handler metrics for monitoring
CREATE TABLE IF NOT EXISTS handler_metrics (
    id SERIAL PRIMARY KEY,
    handler_id TEXT REFERENCES handler_configurations(id),
    conversation_id TEXT NOT NULL,
    event_type TEXT NOT NULL,
    processing_time FLOAT NOT NULL,
    success BOOLEAN NOT NULL,
    error_message TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Conversation transcripts
CREATE TABLE IF NOT EXISTS conversation_transcripts (
    id SERIAL PRIMARY KEY,
    conversation_id TEXT NOT NULL,
    segment_type TEXT NOT NULL,
    content TEXT NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    UNIQUE (conversation_id, created_at)
);

-- Conversation summaries
CREATE TABLE IF NOT EXISTS conversation_summaries (
    id SERIAL PRIMARY KEY,
    conversation_id TEXT NOT NULL,
    summary_text TEXT NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    UNIQUE (conversation_id, created_at)
);

-- Handler state tracking
CREATE TABLE IF NOT EXISTS handler_states (
    id SERIAL PRIMARY KEY,
    handler_id TEXT REFERENCES handler_configurations(id),
    conversation_id TEXT NOT NULL,
    state_json TEXT NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    UNIQUE (handler_id, conversation_id)
);

-- Create indexes
CREATE INDEX IF NOT EXISTS idx_handler_metrics_conversation 
ON handler_metrics (conversation_id);

CREATE INDEX IF NOT EXISTS idx_transcripts_conversation 
ON conversation_transcripts (conversation_id);

CREATE INDEX IF NOT EXISTS idx_summaries_conversation 
ON conversation_summaries (conversation_id);
```
