2. Implement logic to transition from call completion to processing:
   ```go
   // handleCallCompletion processes a completed call
   func (app *application) handleCallCompletion(w http.ResponseWriter, r *http.Request) {
       // Get call details
       callSID := r.FormValue("CallSid")
       
       // Update appointment status
       err := app.appointments.MarkCallComplete(callSID)
       if err != nil {
           app.serverError(w, err)
           return
       }
       
       // Queue transcript for processing
       appointment, err := app.appointments.GetByCallSID(callSID)
       if err != nil {
           app.serverError(w, err)
           return
       }
       
       // Initialize processing status
       err = app.processingStatus.UpdateProcessingStatus(appointment.ID, "queued", nil)
       if err != nil {
           app.serverError(w, err)
           return
       }
       
       // Queue the actual processing job
       err = app.jobs.QueueTranscriptProcessing(appointment.ID)
       if err != nil {
           app.serverError(w, err)
           return
       }
       
       // Return success response
       w.WriteHeader(http.StatusOK)
   }
   ```
